function data_plot_app()
loadedFiles = {};
%% MAIN WINDOW
fig = uifigure('Name','Data Analysis and Plot Builder',...
               'Position',[100 100 1000 600]);

tg = uitabgroup(fig,'Units','normalized','Position',[0 0 1 1]);

tabImport = uitab(tg,'Title','Data Import');
tabPlot   = uitab(tg,'Title','Plot Builder');

%% PLOT BUILDER TAB

plotGrid = uigridlayout(tabPlot,[1 3]);
plotGrid.ColumnWidth = {250,250,'1x'};

%% SIGNAL PANEL

signalPanel = uipanel(plotGrid,'Title','Signals');
signalPanel.Layout.Column = 1;

signalGrid = uigridlayout(signalPanel,[4 1]);

uilabel(signalGrid,'Text','X Axis');

xList = uilistbox(signalGrid);

uilabel(signalGrid,'Text','Y Signals');

yList = uilistbox(signalGrid,'Multiselect','on');

%% PLOT SETTINGS PANEL


settingsPanel = uipanel(plotGrid,'Title','Plot Settings');
settingsPanel.Layout.Column = 2;

settingsGrid = uigridlayout(settingsPanel,[6 2]);



uilabel(settingsGrid,'Text','Plot Type');
plotType = uidropdown(settingsGrid,'Items',{'plot','scatter','stairs'});

uilabel(settingsGrid,'Text','Subplot Row');
subplotRow = uieditfield(settingsGrid,'numeric','Value',1);

uilabel(settingsGrid,'Text','Subplot Column');
subplotCol = uieditfield(settingsGrid,'numeric','Value',1);

legendCheck = uicheckbox(settingsGrid,'Text','Legend','Value',true);
gridCheck = uicheckbox(settingsGrid,'Text','Grid','Value',true);

uilabel(settingsGrid,'Text','Figure Number');
figNum = uieditfield(settingsGrid,'numeric','Value',1);

uilabel(settingsGrid,'Text','subplot index');
subplotindex = uieditfield(settingsGrid,'numeric','Value',1);

uilabel(settingsGrid,'Text','Color')
colorDrop = uidropdown(settingsGrid,...
    'Items',{'auto','r','g','b','k','m','c','y'},...
    'Value','auto');

%% PLOT LIST PANEL

plotPanel = uipanel(plotGrid,'Title','Plot List');
plotPanel.Layout.Column = 3;

plotListGrid = uigridlayout(plotPanel,[5 1]);

plotList = uilistbox(plotListGrid);

btnAddPlot = uibutton(plotListGrid,'Text','Add Plot',...
    'ButtonPushedFcn',@addPlot);

btnPreview = uibutton(plotListGrid,'Text','Preview Plot',...
    'ButtonPushedFcn',@previewPlot);

btnScript = uibutton(plotListGrid,'Text','Generate MATLAB Script',...
    'ButtonPushedFcn',@generateScript);

btnClear = uibutton(plotListGrid,'Text','Clear Plot List',...
    'ButtonPushedFcn',@clearPlots);

%% GRID LAYOUT FOR IMPORT TAB
 igrid = uigridlayout(tabImport,[3 1]);
 igrid.RowHeight={40,40,'1x'};
% grid.ColumnWidtht={'1x'};

%% ROW 1: LOAD BUTTON
row1 = uigridlayout(igrid,[1 3]);

btnLoad = uibutton(row1,...
    'Text','Load MAT File',...
    'ButtonPushedFcn',@loadMat);

lblSearch = uilabel(row1,'Text','Search');

searchBox = uieditfield(row1,'text',...
    'ValueChangedFcn',@searchVar);

%% ROW 2: GLOBAL TOOLS
row2 = uigridlayout(igrid,[1 3]);

uibutton(row2,'Text','Apply Scale To All',...
    'ButtonPushedFcn',@applyScale);

uibutton(row2,'Text','Apply Offset To All',...
    'ButtonPushedFcn',@applyOffset);

uilabel(row2,'Text','Variable Settings');

%% ROW 3: VARIABLE TABLE
columns = {'Name','Label','Unit','Scale','Offset','Style','Width'};

table = uitable(igrid,...
    'ColumnName',columns,...
    'ColumnEditable',[false true true true true true true]);
table.CellEditCallback = @tableEdit;
% table.Layout.Row=3;
% table.Layout.Column=1;

%% STORAGE
dataVars = struct;
plots = {};

%% LOAD MAT FILE
    function loadMat(~,~)

        [files,path] = uigetfile('*.mat','Select MAT files','MultiSelect','on');

        if isequal(files,0)
            return
        end

        if ischar(files)
            files = {files};
        end

        for k = 1:length(files)
            loadedFiles{end+1} = fullfile(path,files{k});
            S = load(fullfile(path,files{k}));
            names = fieldnames(S);

            for i = 1:length(names)

                name = names{i};
                newName = name;
                c = 1;

                while isfield(dataVars,newName)
                    newName = [name '_' num2str(c)];
                    c = c + 1;
                end

                if isfield(dataVars,newName)

                    % variable already exists → keep properties
                    dataVars.(newName).data = data;

                else

                    % new variable → create properties
                    dataVars.(newName).data = S.(name);
                    dataVars.(newName).label = newName;
                    dataVars.(newName).unit = '';
                    dataVars.(newName).scale = 1;
                    dataVars.(newName).offset = 0;
                    dataVars.(newName).style = '-';
                    dataVars.(newName).width = 1.5;


 

                end

            end
        end

        refreshTable()
        updateSignalLists()

    end

%% REFRESH TABLE
    function refreshTable()

        names = fieldnames(dataVars);
        tableData = cell(length(names),7);

        for i = 1:length(names)

            v = dataVars.(names{i});

            tableData(i,:) = {names{i},v.label,v.unit,...
                              v.scale,v.offset,v.style,v.width};
        end

        table.Data = tableData;

    end

%% SEARCH
    function searchVar(src,~)

        keyword = lower(src.Value);
        names = fieldnames(dataVars);

        tableData = {};
        idx = 1;

        for i = 1:length(names)

            if contains(lower(names{i}),keyword)

                v = dataVars.(names{i});

                tableData(idx,:) = {names{i},v.label,v.unit,...
                                    v.scale,v.offset,v.style,v.width};

                idx = idx + 1;

            end
        end

        table.Data = tableData;

    end

%% APPLY SCALE
    function applyScale(~,~)

saveTableToStruct()

if isempty(table.Data)
    return
end

scale = table.Data{1,4};

names = fieldnames(dataVars);

for i = 1:length(names)
    dataVars.(names{i}).scale = scale;
end

refreshTable()

end

%% APPLY OFFSET
    function applyOffset(~,~)

saveTableToStruct()

if isempty(table.Data)
    return
end

offset = table.Data{1,5};

names = fieldnames(dataVars);

for i = 1:length(names)
    dataVars.(names{i}).offset = offset;
end

refreshTable()

end


function saveTableToStruct()

names = fieldnames(dataVars);

for i = 1:size(table.Data,1)

    name = table.Data{i,1};

    dataVars.(name).label  = table.Data{i,2};
    dataVars.(name).unit   = table.Data{i,3};
    dataVars.(name).scale  = table.Data{i,4};
    dataVars.(name).offset = table.Data{i,5};
    dataVars.(name).style  = table.Data{i,6};
    dataVars.(name).width  = table.Data{i,7};

end
end

    function addPlot(~,~)

        x = xList.Value;
        y = yList.Value;

        if isempty(x) || isempty(y)
            return
        end

        entry = struct;

        entry.x = x;
        entry.y = y;

        entry.type = plotType.Value;

        entry.row = subplotRow.Value;
        entry.col = subplotCol.Value;
        entry.pos = subplotindex.Value;

        entry.figNum = figNum.Value;

        entry.color = colorDrop.Value;   % NEW (dropdown for color)

        plots{end+1} = entry;

        plotList.Items{end+1} = [x ' vs ' strjoin(y,', ')];

    end

function previewPlot(~,~)

if isempty(plots)
    return
end

usedSubplots = containers.Map;

for k = 1:length(plots)

    p = plots{k};

    figNum1 = p.figNum;
    rows = p.row;
    cols = p.col;
    pos  = p.pos;

    key = sprintf('%d_%d_%d_%d',figNum1,rows,cols,pos);

    figure(figNum1)

    if isKey(usedSubplots,key)
        subplot(rows,cols,pos)
        hold on
    else
        subplot(rows,cols,pos)
        hold on
        usedSubplots(key) = true;
    end

    % X variable
    xVar = p.x;
    xdata = dataVars.(xVar).data;
    xdata = xdata * dataVars.(xVar).scale + dataVars.(xVar).offset;

    % loop through Y variables
    for i = 1:length(p.y)

        yVar = p.y{i};

        ydata = dataVars.(yVar).data;
        ydata = ydata * dataVars.(yVar).scale + dataVars.(yVar).offset;

        style = dataVars.(yVar).style;
        width = dataVars.(yVar).width;
        label = dataVars.(yVar).label;

        % COLOR handling
        if strcmp(p.color,'auto')
            plot(xdata,ydata,style,...
                'LineWidth',width,...
                'DisplayName',label)
        else
            plot(xdata,ydata,style,...
                'LineWidth',width,...
                'Color',p.color,...
                'DisplayName',label)
        end

    end

    xlabel([dataVars.(xVar).label ' ' dataVars.(xVar).unit])

    if gridCheck.Value
        grid on
    end

end

if legendCheck.Value
    legend show
end

end

function generateScript(~,~)

fid = fopen('generated_plot.m','w');

fprintf(fid,'%% AUTO GENERATED SCRIPT\n');
fprintf(fid,'clear\n');
fprintf(fid,'close all\n\n');

%% LOAD DATA
fprintf(fid,'%% Load MAT files safely\n');

fprintf(fid,'dataVars = struct;\n');
fprintf(fid,'\n');

for i = 1:length(loadedFiles)

    fprintf(fid,'S = load(''%s'');\n',loadedFiles{i});
    fprintf(fid,'names = fieldnames(S);\n');
    fprintf(fid,'\n');
    fprintf(fid,'for iVar = 1:length(names)\n');
    fprintf(fid,'    name = names{iVar};\n');
    fprintf(fid,'    newName = name;\n');
    fprintf(fid,'    c = 1;\n');
    fprintf(fid,'\n');
    fprintf(fid,'    while isfield(dataVars,newName)\n');
    fprintf(fid,'        newName = [name ''_'' num2str(c)];\n');
    fprintf(fid,'        c = c + 1;\n');
    fprintf(fid,'    end\n');
    fprintf(fid,'\n');
    fprintf(fid,'    dataVars.(newName) = S.(name);\n');
    fprintf(fid,'end\n\n');

end


%% APPLY SCALE AND OFFSET
fprintf(fid,'%% Apply scale and offset\n');

names = fieldnames(dataVars);

for i = 1:length(names)

    n = names{i};

    scale  = dataVars.(n).scale;
    offset = dataVars.(n).offset;

    fprintf(fid,'%s = %f*dataVars.%s + %f;\n',n,scale,n,offset);

end

fprintf(fid,'\n');


%% PLOT SECTION
fprintf(fid,'%% Plot section\n');

currentFig = -1;

for k = 1:length(plots)

    p = plots{k};

    if p.figNum ~= currentFig
        fprintf(fid,'figure(%d)\n',p.figNum);
        currentFig = p.figNum;
    end

    fprintf(fid,'subplot(%d,%d,%d)\n',p.row,p.col,p.pos);
    fprintf(fid,'hold on\n');

    for j = 1:length(p.y)

        yVar = p.y{j};
        xVar = p.x;

        style = dataVars.(yVar).style;
        width = dataVars.(yVar).width;
        label = dataVars.(yVar).label;
        color = p.color;

        if strcmp(color,'auto')

            fprintf(fid,...
            'plot(%s,%s,''%s'',''LineWidth'',%f,''DisplayName'',''%s'')\n',...
            xVar,yVar,style,width,label);

        else

            fprintf(fid,...
            'plot(%s,%s,''%s'',''Color'',''%s'',''LineWidth'',%f,''DisplayName'',''%s'')\n',...
            xVar,yVar,style,color,width,label);

        end

    end

    fprintf(fid,'grid on\n\n');

end

fprintf(fid,'legend show\n');

fclose(fid);

disp('generated_plot.m created');

end

function clearPlots(~,~)

plots = {};
plotList.Items = {};

end

function updateSignalLists()

names = fieldnames(dataVars);

xList.Items = names;
yList.Items = names;

end

function tableEdit(src,event)

row = event.Indices(1);
name = src.Data{row,1};

dataVars.(name).label  = src.Data{row,2};
dataVars.(name).unit   = src.Data{row,3};
dataVars.(name).scale  = src.Data{row,4};
dataVars.(name).offset = src.Data{row,5};
dataVars.(name).style  = src.Data{row,6};
dataVars.(name).width  = src.Data{row,7};

end
end
