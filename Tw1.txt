        %% Core Engines
        function safeExecute(obj, functionHandle, startMsg)
            try
                if ~isempty(startMsg), obj.logMessage('ACTION', startMsg); end
                functionHandle();
            catch ME
                errMsg = sprintf('%s (Line %d)', ME.message, ME.stack(1).line);
                obj.logMessage('ERROR', errMsg);
                uialert(obj.UI.Fig, errMsg, 'Runtime Error');
            end
        end
        
        function logMessage(obj, level, msg)
            txt = sprintf('[%s] [%s] %s', datestr(now, 'HH:MM:SS'), level, msg);
            obj.UI.Console.Value{end+1} = txt;
            scroll(obj.UI.Console, 'bottom');
        end

        function str = fixLaTeXColors(~, str)
            if isempty(str), return; end
            map = {'\color{r}', '\color{red}'; '\color{g}', '\color{green}'; 
                   '\color{b}', '\color{blue}'; '\color{c}', '\color{cyan}'; 
                   '\color{m}', '\color{magenta}'; '\color{y}', '\color{yellow}'; 
                   '\color{k}', '\color{black}'};
            for i = 1:size(map, 1)
                str = strrep(str, map{i, 1}, map{i, 2});
            end
        end

        function actionLoadMat(obj)
            [file,path] = uigetfile('*.mat', 'Select MAT Files', 'MultiSelect','on');
            if isequal(file,0), return; end
            if ischar(file), file = {file}; end
            
            prefixStr = inputdlg('Enter variable prefix (blank for none):', 'Prefix', [1 40], {''});
            if isempty(prefixStr), pref = ''; else, pref = prefixStr{1}; end
            
            for k = 1:numel(file)
                fullFile = fullfile(path,file{k});
                obj.LoadedFile = file{k};
                obj.LoadedFiles{end+1} = fullFile;
                rawStruct = load(fullFile);
                obj.parseStructRecursive(rawStruct, '', pref);
            end
            obj.actionRefreshTable(); obj.syncUILists();
        end
        
        function parseStructRecursive(obj, inStruct, prefixPath, namePrefix)
            fn = fieldnames(inStruct);
            for i = 1:length(fn)
                val = inStruct.(fn{i});
                fullPath = fn{i};
                if ~isempty(prefixPath), fullPath = [prefixPath '.' fn{i}]; end
                
                if isstruct(val)
                    obj.parseStructRecursive(val, fullPath, namePrefix);
                elseif isnumeric(val) || islogical(val)
                    safeName = [namePrefix matlab.lang.makeValidName(fullPath)];
                    safeName = obj.makeUniqueName(safeName);
                    
                    meta = struct('data', double(val(:)), 'label', safeName, 'unit', '', ...
                                  'offset', 0, 'transpose', false, 'isDerived', false, ...
                                  'expr', '', 'cache', [], 'delay', 0);
                    obj.DataVars.(safeName) = meta;
                end
            end
        end
        
        function safeName = makeUniqueName(obj, safeName)
            baseName = safeName; count = 1;
            while isfield(obj.DataVars, safeName)
                safeName = matlab.lang.makeValidName(sprintf('%s_%d', baseName, count));
                count = count + 1;
            end
        end
        
        function sig = fetchSignal(obj, key)
            if isempty(key) || ~(ischar(key) || isstring(key)) || ~isfield(obj.DataVars, key), sig = []; return; end
            meta = obj.DataVars.(key);
            if meta.isDerived
                sig = obj.evaluateMathSafely(meta.expr);
                obj.DataVars.(key).cache = sig;
                return;
            end
            if ~isempty(meta.cache), sig = meta.cache; return; end
            sig = meta.data + meta.offset;
            obj.DataVars.(key).cache = sig;
        end
        
        function out = evaluateMathSafely(obj, expr)
            vars = unique(regexp(expr, '[a-zA-Z]\w*', 'match'));
            for i = 1:length(vars)
                if isfield(obj.DataVars, vars{i})
                    eval(sprintf('%s = obj.fetchSignal(''%s'');', vars{i}, vars{i}));
                end
            end
            out = eval(expr);
        end
        
        %% State Validation & Logic Libraries
        function actionValidateState(obj)
            reqVars = {}; missingVars = {};
            
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                if strcmp(p.type, 'plot')
                    reqVars = [reqVars, p.x, p.y];
                    if p.is3D, reqVars = [reqVars, p.z]; end
                    if p.is4D && ~isempty(p.sVar), reqVars{end+1} = p.sVar; reqVars{end+1} = p.cVar; end
                end
            end
            
            dKeys = fieldnames(obj.SavedDerivedLogic);
            for i=1:length(dKeys)
                reqVars = [reqVars, unique(regexp(obj.SavedDerivedLogic.(dKeys{i}).expr, '[a-zA-Z]\w*', 'match'))];
            end
            
            eKeys = fieldnames(obj.SavedEventLogic);
            for i=1:length(eKeys)
                reqVars = [reqVars, unique(regexp(obj.SavedEventLogic.(eKeys{i}).expr, '[a-zA-Z]\w*', 'match'))];
            end
            
            reqVars = unique(reqVars);
            for i = 1:length(reqVars)
                if ~isempty(reqVars{i}) && ~isfield(obj.DataVars, reqVars{i}) && ~isfield(obj.SavedDerivedLogic, reqVars{i})
                    missingVars{end+1} = reqVars{i};
                end
            end
            
            if ~isempty(missingVars)
                uialert(obj.UI.Fig, sprintf('Missing base variables:\n%s', strjoin(missingVars, '\n')), 'Validation Failed', 'Icon', 'error');
                return;
            end
            
            for i=1:length(dKeys)
                if ~isfield(obj.DataVars, dKeys{i})
                    s = obj.SavedDerivedLogic.(dKeys{i});
                    out = obj.evaluateMathSafely(s.expr);
                    obj.DataVars.(dKeys{i}) = struct('data', out(:), 'label', s.name, 'unit', s.unit, ...
                        'offset', 0, 'transpose', false, 'isDerived', true, 'expr', s.expr, 'cache', out(:), 'delay', 0);
                end
            end
            
            obj.actionRefreshTable(); obj.syncUILists();
            uialert(obj.UI.Fig, 'Validation Passed. Derived signals auto-recreated if missing.', 'Valid', 'Icon', 'success');
        end

        function actionSaveState(obj)
            [file, path] = uiputfile('*.mat', 'Save State'); if file == 0, return; end
            
            lightDataVars = struct();
            fn = fieldnames(obj.DataVars);

        function syncManifest(obj)
            delete(obj.UI.plotTree.Children);
            if ~isempty(obj.Plots)
                numPlots = length(obj.Plots);
                gList = zeros(numPlots, 1);
                fList = zeros(numPlots, 1);
                pList = zeros(numPlots, 1);
                for i = 1:numPlots
                    gList(i) = obj.Plots{i}.group;
                    fList(i) = obj.Plots{i}.f;
                    pList(i) = obj.Plots{i}.p;
                end
                
                T = table(gList, fList, pList, (1:numPlots)', 'VariableNames', {'group', 'f', 'p', 'idx'});
                T = sortrows(T, {'group', 'f', 'p'});
                obj.Plots = obj.Plots(T.idx);
            end
            
            grpMap = containers.Map('KeyType', 'double', 'ValueType', 'any');
            for p = 1:length(obj.Plots)
                g = obj.Plots{p}.group;
                if isKey(grpMap, g), grpMap(g) = [grpMap(g) {obj.Plots{p}}]; else, grpMap(g) = {obj.Plots{p}}; end
            end
            
            gKeys = sort(cell2mat(grpMap.keys()));
            for k = 1:length(gKeys)
                gNode = uitreenode(obj.UI.plotTree, 'Text', sprintf('Group %d', gKeys(k)), 'NodeData', struct('type', 'grp', 'id', gKeys(k)));
                
                figMap = containers.Map('KeyType', 'double', 'ValueType', 'any');
                gList = grpMap(gKeys(k));
                for p = 1:length(gList)
                    f = gList{p}.f;
                    if isKey(figMap, f), figMap(f) = [figMap(f) {gList{p}}]; else, figMap(f) = {gList{p}}; end
                end
                
                fKeys = sort(cell2mat(figMap.keys()));
                for i = 1:length(fKeys)
                    fNode = uitreenode(gNode, 'Text', sprintf('Figure %d', fKeys(i)), 'NodeData', struct('type', 'fig', 'id', fKeys(i)));
                    pList = figMap(fKeys(i));
                    for j = 1:length(pList)
                        pt = pList{j};
                        if strcmp(pt.type, 'plot')
                            pName = sprintf('Sub %d: Y=%s', pt.p, strjoin(pt.y, ','));
                        else
                            pName = sprintf('Sub %d: [%s] %s', pt.p, upper(pt.type), pt.name);
                        end
                        uitreenode(fNode, 'Text', pName, 'NodeData', struct('type', pt.type, 'id', pt.id));
                    end
                    expand(fNode);
                end
                expand(gNode);
            end
        end

        %% Master Render Engine
        function renderAll(obj, targetFig, isOverlay, targetGroups, targetPlots)
            if nargin < 3, isOverlay = false; end
            if nargin < 4, targetGroups = []; end
            if nargin < 5, targetPlots = {}; end
            
            allFigs = unique(cellfun(@(x)x.f, obj.Plots));
            if ~isempty(targetFig), allFigs = intersect(allFigs, targetFig); end
            
            if ~isOverlay, for i = 1:length(allFigs), figure(allFigs(i)); clf(figure(allFigs(i))); end; end
            
            distinctColors = lines(15);
            sgTitlesMap = containers.Map('KeyType', 'double', 'ValueType', 'any');
            axMap = containers.Map('KeyType', 'double', 'ValueType', 'any');
            
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                if ~isempty(targetFig) && ~ismember(p.f, targetFig), continue; end
                if ~isOverlay && (~isempty(targetGroups) || ~isempty(targetPlots))
                    if ~ismember(p.group, targetGroups) && ~any(strcmp(p.id, targetPlots)), continue; end
                end
                
                ax = subplot(p.r, p.c, p.p, 'Parent', figure(p.f)); hold(ax, 'on');
                
                if isfield(p, 'sgTitle') && ~isempty(p.sgTitle)
                    sgTitlesMap(p.f) = {p.sgTitle, p.sgBTitle};
                end
                
                if strcmp(p.type, 'event_line')
                    xV = obj.fetchSignal(p.refX);
                    for t=1:length(p.onIdx)
                        if p.onIdx(t) <= length(xV), xline(ax, xV(p.onIdx(t)), '-', 'Color', p.color, 'LineWidth', 1.5, 'HandleVisibility', 'off'); end
                        if p.offIdx(t) <= length(xV), xline(ax, xV(p.offIdx(t)), '--', 'Color', p.color, 'LineWidth', 1.5, 'HandleVisibility', 'off'); end
                    end
                    continue;
                elseif strcmp(p.type, 'event_marker')
                    xV = obj.fetchSignal(p.xLine); yV = obj.fetchSignal(p.yLine);
                    if isempty(xV) || isempty(yV), continue; end 
                    
                    validIdx = p.onIdx(p.onIdx <= length(xV) & p.onIdx <= length(yV));
                    if ~isempty(validIdx)
                        if isfield(p, 'zLine') && ~isempty(p.zLine)
                            zV = obj.fetchSignal(p.zLine);
                            validIdx = validIdx(validIdx <= length(zV));
                            plot3(ax, xV(validIdx), yV(validIdx), zV(validIdx), 'o', 'Color', p.color, 'MarkerFaceColor', p.color, 'MarkerSize', 8, 'LineWidth', 1, 'HandleVisibility', 'off', 'Tag', 'EventMarker', 'DisplayName', p.name);
                        else
                            plot(ax, xV(validIdx), yV(validIdx), 'o', 'Color', p.color, 'MarkerFaceColor', p.color, 'MarkerSize', 8, 'LineWidth', 1, 'HandleVisibility', 'off', 'Tag', 'EventMarker', 'DisplayName', p.name);
                        end
                    end
                    continue;
                end
                
                autoTitleChunks = {}; firstValidX = '';
                if ~isempty(p.x), firstValidX = p.x{1}; end
                
                for j = 1:length(p.y)
                    if p.linkIDs(j) > 0
                        if isKey(axMap, p.linkIDs(j)), axMap(p.linkIDs(j)) = [axMap(p.linkIDs(j)), ax];
                        else, axMap(p.linkIDs(j)) = ax; end
                    end
                end
                
                if p.is4D
                    xV = obj.fetchSignal(p.x{1}); yV = obj.fetchSignal(p.y{1}) * p.yScales(1);
                    cV = obj.fetchSignal(p.sVar); len = min([length(xV), length(yV), length(cV)]);
                    [cleanLS, cleanMarker] = obj.parseLineStyle(p.yStyles{1});
                    
                    validC = cV(1:len); uStates = unique(validC(~isnan(validC)));
                    numStates = length(uStates); if numStates == 0, numStates = 1; end
                    
                    colormap(ax, lines(numStates));
                    cMapped = zeros(size(validC)); for k = 1:numStates, cMapped(validC == uStates(k)) = k; end
                    caxis(ax, [1, numStates]); cb = colorbar(ax); cb.Ticks = 1:numStates;
                    
                    if isfield(p, 'tickLabels') && length(p.tickLabels) == numStates, cb.TickLabels = p.tickLabels;
                    else, cb.TickLabels = cellfun(@num2str, num2cell(uStates), 'UniformOutput', false); end
                    
                    dispName = strrep(obj.DataVars.(p.y{1}).label, '_', '\_');
                    if p.is3D
                        zV = obj.fetchSignal(p.z{1});
                        surface(ax, [xV(1:len) xV(1:len)], [yV(1:len) yV(1:len)], [zV(1:len) zV(1:len)], [cMapped(1:len)' cMapped(1:len)'], ...
                            'FaceColor', 'none', 'EdgeColor', 'flat', 'LineWidth', p.yWidths(1), 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dispName);
                    else
                        surface(ax, [xV(1:len) xV(1:len)], [yV(1:len) yV(1:len)], zeros(len, 2), [cMapped(1:len) cMapped(1:len)], ...
                            'FaceColor', 'none', 'EdgeColor', 'flat', 'LineWidth', p.yWidths(1), 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dispName);
                    end
                else
                    for j = 1:length(p.y)
                        xV = obj.fetchSignal(p.x{j}); yV = obj.fetchSignal(p.y{j}) * p.yScales(j);
                        len = min(length(xV), length(yV));
                        [cleanLS, cleanMarker] = obj.parseLineStyle(p.yStyles{j});
                        
                        dispName = strrep(obj.DataVars.(p.y{j}).label, '_', '\_');
                        
                        if length(yV) == 1 && strcmp(p.yStyles{j}, 'yline')
                            yline(ax, yV, 'Color', p.yColors{j}, 'LineWidth', p.yWidths(j), 'DisplayName', dispName); continue;
                        elseif length(yV) == 1 && strcmp(p.yStyles{j}, 'xline')
                            xline(ax, yV, 'Color', p.yColors{j}, 'LineWidth', p.yWidths(j), 'DisplayName', dispName); continue;
                        end
                        
                        visMode = 'off'; if p.yLeg(j), visMode = 'on'; end
                        args = {'LineWidth', p.yWidths(j), 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dispName, 'HandleVisibility', visMode};
                        
                        cRGB = p.yColors{j};
                        if strcmp(cRGB, 'auto')
                            cRGB = distinctColors(mod(length(findall(ax, 'Type', 'line')) + j - 1, 15) + 1, :);
                            args = [args, 'Color', cRGB];
                            if ~strcmp(cleanMarker, 'none'), args = [args, 'MarkerFaceColor', cRGB]; end
                            if p.yLeg(j), autoTitleChunks{end+1} = sprintf('\\color[rgb]{%.3f,%.3f,%.3f}%s', cRGB(1), cRGB(2), cRGB(3), dispName); end
                        else
                            args = [args, 'Color', cRGB];
                            if ~strcmp(cleanMarker, 'none'), args = [args, 'MarkerFaceColor', cRGB]; end
                            if p.yLeg(j), autoTitleChunks{end+1} = sprintf('%s%s', obj.fixLaTeXColors(['\color{' cRGB '}']), dispName); end
                        end
                        
                        if p.is3D, zV = obj.fetchSignal(p.z{j}); plot3(ax, xV(1:len), yV(1:len), zV(1:len), args{:});
                        else, plot(ax, xV(1:len), yV(1:len), args{:}); end
                    end
                end
                
                plotTitleStr = obj.fixLaTeXColors(p.title);
                if isempty(p.title)
                    if ~isempty(autoTitleChunks), plotTitleStr = strjoin(autoTitleChunks, ' \color{black}/ '); end
                    if ~isempty(p.yLabels{1}), plotTitleStr = [plotTitleStr ' / ' p.yLabels{1}]; end
                end
                if ~isempty(plotTitleStr), title(ax, plotTitleStr, 'Interpreter', 'tex'); end
                
                if ~isempty(firstValidX), xlabel(ax, strrep(obj.DataVars.(firstValidX).label, '_', '\_'), 'Interpreter', 'none'); end
                if p.is3D && ~isempty(p.z{1}), zlabel(ax, strrep(obj.DataVars.(p.z{1}).label, '_', '\_'), 'Interpreter', 'none'); end
                if p.grid, grid(ax, 'on'); end; if p.leg, legend(ax, 'Location', 'best'); end; if p.is3D, view(ax, 3); end
            end
            
            fKs = sgTitlesMap.keys();
            for f = 1:length(fKs)
                strs = sgTitlesMap(fKs{f});
                sgtitle(figure(fKs{f}), {strs{1}, strs{2}}, 'Interpreter', 'tex', 'FontWeight', 'bold');
            end
            
            linkIds = axMap.keys(); for i = 1:length(linkIds), if length(axMap(linkIds{i})) > 1, linkaxes(axMap(linkIds{i}), 'x'); end; end
        end
        
        %% Advanced Event Engine & Markers
        function actionAddEvent(obj, typeStr)
            obj.actionSaveEventLogic();
            expr = char(strtrim(obj.UI.evLogic.Value));
            mask = obj.evaluateMathSafely(expr);
            mask = double(mask(:)); mask(~isfinite(mask)) = 0; mask(mask ~= 0) = 1;
            
            d = diff([0; mask; 0]); startIdx = find(d == 1); stopIdx = find(d == -1) - 1;
            idxFind = find(mask == 1);
            if isempty(idxFind), uialert(obj.UI.Fig, 'Logic expression is never true.', 'Notice'); return; end
            
            fn = fieldnames(obj.DataVars);
            [indx, tf] = listdlg('PromptString', 'Select Time/X Reference Channel for logic evaluation:', 'SelectionMode', 'single', 'ListString', fn);
            if ~tf, return; end
            
            t = obj.fetchSignal(fn{indx}); t = t(:);
            e = struct('id', char(javaMethod('randomUUID','java.util.UUID')), 'name', char(obj.UI.evName.Value), ...
                'type', typeStr, ...
                'expr', expr, 'refX', fn{indx}, ...
                'onT', t(startIdx), 'offT', t(stopIdx), 'onIdx', startIdx, 'offIdx', stopIdx, ...
                'fig', obj.UI.evFig.Value, 'c', char(obj.UI.evColor.Value));
            
            obj.Events{end+1} = e; obj.syncEventList();
            
            posStr = char(obj.UI.evPos.Value);
            posArray = str2num(posStr);
            if isempty(posArray), posArray = 1; end
            
            if strcmp(typeStr, 'line')
                for posIdx = 1:length(posArray)
                    p = struct('id', char(javaMethod('randomUUID','java.util.UUID')), 'type', 'event_line', 'name', e.name, 'group', obj.UI.pltGroup.Value, 'f', e.fig, ...
                               'r', obj.UI.pltRow.Value, 'c', obj.UI.pltCol.Value, 'p', posArray(posIdx), 'color', e.c, ...
                               'onIdx', e.onIdx, 'offIdx', e.offIdx, 'refX', e.refX);
                    obj.Plots{end+1} = p; 
                end
                obj.syncManifest(); obj.renderAll([e.fig], true, [], {});
            elseif strcmp(typeStr, 'marker')
                fNum = e.fig; if ~ishandle(fNum), return; end
                axList = findall(figure(fNum), 'Type', 'axes', '-not', 'Tag', 'legend');
                
                for posIdx = 1:length(posArray)
                    if posArray(posIdx) > length(axList), continue; end
                    ax = axList(end - posArray(posIdx) + 1);
                    curves = findobj(ax, 'Type', 'line', '-or', 'Type', 'surface');
                    if isempty(curves), continue; end
                    
                    names = cell(length(curves), 1);
                    for i=1:length(curves)
                        if ~isempty(curves(i).DisplayName), names{i} = curves(i).DisplayName; else, names{i} = sprintf('Curve (%d)', i); end
                    end
                    [cIdx, tf] = listdlg('PromptString', sprintf('Select target line for markers on Subplot %d:', posArray(posIdx)), 'SelectionMode', 'single', 'ListString', names);
                    if ~tf, continue; end
                    
                    targetYKey = ''; targetXKey = fn{indx}; targetZKey = '';
                    for k=1:length(obj.Plots)
                        if obj.Plots{k}.f == e.fig && obj.Plots{k}.p == posArray(posIdx) && strcmp(obj.Plots{k}.type, 'plot')
                            for yIdx = 1:length(obj.Plots{k}.y)
                                if strcmp(strrep(obj.DataVars.(obj.Plots{k}.y{yIdx}).label, '_', '\_'), curves(cIdx).DisplayName)
                                    targetYKey = obj.Plots{k}.y{yIdx};
                                    targetXKey = obj.Plots{k}.x{yIdx};
                                    if obj.Plots{k}.is3D, targetZKey = obj.Plots{k}.z{yIdx}; end
                                    break;
                                end
                            end
                        end
                    end
                    
                    if isempty(targetYKey)
                        for i=1:length(fn)
                            if strcmp(strrep(obj.DataVars.(fn{i}).label, '_', '\_'), curves(cIdx).DisplayName), targetYKey = fn{i}; break; end
                        end
                    end
                    
                    if isempty(targetYKey)
                        uialert(obj.UI.Fig, 'Could not map selected curve back to a known dataset variable.', 'Mapping Error'); return; 
                    end 
                    
                    p = struct('id', char(javaMethod('randomUUID','java.util.UUID')), 'type', 'event_marker', 'name', e.name, ...
                               'group', obj.UI.pltGroup.Value, 'f', e.fig, 'r', obj.UI.pltRow.Value, 'c', obj.UI.pltCol.Value, 'p', posArray(posIdx), 'color', e.c, ...
                               'onIdx', idxFind, 'xLine', targetXKey, 'yLine', targetYKey, 'zLine', targetZKey);
                    obj.Plots{end+1} = p; 
                end
                obj.syncManifest(); obj.renderAll([e.fig], true, [], {});
            end
        end
        
        function actionAddCustomDatatip(obj)
            figNum = obj.UI.evFig.Value; posStr = char(obj.UI.evPos.Value);
            posArray = str2num(posStr); if isempty(posArray), posArray = 1; end
            if ~ishandle(figNum), uialert(obj.UI.Fig, 'Target Figure is not open.', 'Error'); return; end
            
            axList = findall(figure(figNum), 'Type', 'axes', '-not', 'Tag', 'legend');
            
            for posIdx = 1:length(posArray)
                if posArray(posIdx) > length(axList), continue; end
                ax = axList(end - posArray(posIdx) + 1);
                
                curves = findobj(ax, 'Type', 'line', 'Tag', 'EventMarker'); 
                if isempty(curves), continue; end
                
                [cIdx, tf] = listdlg('PromptString', sprintf('Select Marker on Subplot %d:', posArray(posIdx)), 'SelectionMode', 'single', 'ListString', arrayfun(@(x) sprintf('Markers (%d points) - %s', numel(x.XData), x.DisplayName), curves, 'Uniform', false));
                if ~tf, continue; end
                targetCurve = curves(cIdx);
                
                evName = targetCurve.DisplayName;
                if isempty(evName), evName = char(obj.UI.evName.Value); end
                
                xl = char(obj.UI.evXName.Value);
                if isempty(xl)
                    xl = ax.XLabel.String; 
                    if isempty(xl), xl = 'XData'; end
                end
                
                yl = char(obj.UI.evYName.Value);
                if isempty(yl)
                    yl = 'YData';
                end
                
                zl = char(obj.UI.evZName.Value);
                hasZ = isprop(targetCurve, 'ZData') && ~isempty(targetCurve.ZData) && any(targetCurve.ZData(:) ~= 0);
                if isempty(zl) && hasZ
                    zl = ax.ZLabel.String; 
                    if isempty(zl), zl = 'ZData'; end
                end
                
                dtt = targetCurve.DataTipTemplate;
                dtt.DataTipRows(1).Label = xl; dtt.DataTipRows(2).Label = yl;
                if hasZ && numel(dtt.DataTipRows) >= 3
                    dtt.DataTipRows(3).Label = zl;
                    while numel(dtt.DataTipRows) > 3, dtt.DataTipRows(end) = []; end
                else
                    while numel(dtt.DataTipRows) > 2, dtt.DataTipRows(end) = []; end
                end
                
                evtText = repmat({evName}, numel(targetCurve.XData), 1);
                dtt.DataTipRows(end+1) = dataTipTextRow('Event:', evtText);
                
                for i = 1:numel(targetCurve.XData)
                    if hasZ
                        datatip(targetCurve, targetCurve.XData(i), targetCurve.YData(i), targetCurve.ZData(i));
                    else
                        datatip(targetCurve, targetCurve.XData(i), targetCurve.YData(i));
                    end
                end
            end
        end
        
        function syncEventList(obj)
            items = {}; ids = {};
            for i = 1:length(obj.Events)
                e = obj.Events{i};
                if isfield(e, 'type') && strcmp(e.type, 'marker')
                    continue;
                end
                for j = 1:length(e.onT)
                    items{end+1} = sprintf('[F%d] %s: Seg %d (%.1fs - %.1fs)', e.fig, e.name, j, e.onT(j), e.offT(j));
                    ids{end+1} = struct('fig', e.fig, 't1', e.onT(j), 't2', e.offT(j), 'eid', e.id, 'segIdx', j);
                end
            end
            if isempty(items)
                obj.UI.evSegList.Enable = 'off'; obj.UI.evSegList.Items = {'No Segments'}; obj.UI.evSegList.ItemsData = {};
            else
                obj.UI.evSegList.Enable = 'on'; obj.UI.evSegList.Items = items; obj.UI.evSegList.ItemsData = ids;
            end
        end

        function actionDeleteSegment(obj)
            seg = obj.UI.evSegList.Value; if isempty(seg) || ~isstruct(seg), return; end
            for i = 1:length(obj.Events)
                if strcmp(obj.Events{i}.id, seg.eid)
                    obj.Events{i}.onT(seg.segIdx) = []; obj.Events{i}.offT(seg.segIdx) = [];
                    obj.Events{i}.onIdx(seg.segIdx) = []; obj.Events{i}.offIdx(seg.segIdx) = [];
                    break;
                end
            end
            obj.syncEventList();
        end

        function actionZoomSegment(obj)
            seg = obj.UI.evSegList.Value; if isempty(seg) || ~isstruct(seg), return; end
            pad = obj.UI.evZoomPad.Value;
            if ishandle(seg.fig)
                axList = findall(figure(seg.fig), 'Type', 'axes', '-not', 'Tag', 'legend');
                for a = 1:length(axList), xlim(axList(a), [seg.t1 - pad, seg.t2 + pad]); end
            end
        end
        
        function actionPlotSegment(obj)
            seg = obj.UI.evSegList.Value; if isempty(seg) || ~isstruct(seg), return; end
            activePlots = {};
            for p=1:length(obj.Plots), if obj.Plots{p}.f == seg.fig && strcmp(obj.Plots{p}.type, 'plot'), activePlots{end+1} = obj.Plots{p}; end; end
            if isempty(activePlots), return; end
            
            f = figure('Name', 'Segment Extracted Plot'); ax = axes(f); hold(ax, 'on');
            for p=1:length(activePlots)
                pt = activePlots{p};
                for y=1:length(pt.y)
                    xV = obj.fetchSignal(pt.x{y});
                    yV = obj.fetchSignal(pt.y{y}) * pt.yScales(y);
                    idx = (xV >= seg.t1) & (xV <= seg.t2);
                    plot(ax, xV(idx), yV(idx), 'LineWidth', pt.yWidths(y), 'DisplayName', pt.y{y});
                end
            end
            legend(ax, 'show', 'Interpreter', 'none'); title(ax, 'Isolated Event Segment'); grid(ax, 'on');
        end

            end
            t = array2table(expData, 'VariableNames', hdr); writetable(t, obj.UI.extFile.Value);
            obj.logMessage('SYSTEM', 'CSV Range Data Subset Export Complete.');
        end

            [file, path] = uiputfile('*.m', 'Save Render Script', 'telemetry_render_v4.m');
            if file == 0, return; end
            fid = fopen(fullfile(path, file), 'w');
            
            fprintf(fid, '%% STANDALONE PROCEDURAL REPRODUCTION CODE\nclearvars; clc;\n\n%% 1. Loading Vectors\n');
            for f = 1:length(obj.LoadedFiles), fprintf(fid,'load(''%s'');\n', strrep(obj.LoadedFiles{f},'''','''''')); end
            
            fprintf(fid, '\n%% 2. Formula Evaluation\n');
            dKeys = fieldnames(obj.SavedDerivedLogic);
            for i=1:length(dKeys)
                s = obj.SavedDerivedLogic.(dKeys{i}); fprintf(fid, '%s = %s;\n', s.name, s.expr);
            end
            
            fprintf(fid, '\n%% 3. Plot Manifest Loop Execution\n');
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                if ~strcmp(p.type, 'plot'), continue; end
                
                fprintf(fid, '\nfigure(%d); ax = subplot(%d, %d, %d); hold(ax, ''on'');\n', p.f, p.r, p.c, p.p);
                
                if p.is4D
                    fprintf(fid, 'cV = %s;\n', p.cVar);
                    fprintf(fid, 'validC = cV(~isnan(cV));\n');
                    fprintf(fid, 'uStates = unique(validC);\n');
                    fprintf(fid, 'numStates = length(uStates); if numStates == 0, numStates = 1; end\n');
                    
                    fprintf(fid, 'colormap(ax, lines(numStates));\n');
                    fprintf(fid, 'cMapped = zeros(size(validC));\n');
                    fprintf(fid, 'for k = 1:numStates, cMapped(validC == uStates(k)) = k; end\n');
                    fprintf(fid, 'caxis(ax, [1, numStates]);\n');
                    
                    if isfield(p, 'sVar') && ~isempty(p.sVar) && ~isempty(p.tickVals)
                        fprintf(fid, 'cb = colorbar(ax);\n');
                        fprintf(fid, 'cb.Ticks = 1:numStates;\n');
                        
                        labelStr = '{';

                            if lblIdx > 1, labelStr = [labelStr ', ']; end

                        end
                        labelStr = [labelStr '}'];
                        fprintf(fid, 'cb.TickLabels = %s;\n', labelStr);
                    end
                    
                    fprintf(fid, 'surface(ax, [%s, %s], [%s, %s], zeros(length(%s),2), [cMapped, cMapped], ''FaceColor'',''none'', ''EdgeColor'',''flat'', ''LineWidth'', %g);\n', ...
                        p.x{1}, p.x{1}, p.y{1}, p.y{1}, p.x{1}, p.yWidths(1));
                else
                    for j = 1:length(p.y)
                        yScale = 1; if isfield(p, 'yScales') && length(p.yScales) >= j, yScale = p.yScales(j); end
                        yStyle = p.lS; if isfield(p, 'yStyles') && length(p.yStyles) >= j, yStyle = p.yStyles{j}; end
                        [cleanLS, cleanMarker] = obj.parseLineStyle(yStyle);
                        
                        fprintf(fid, 'yV = %s * %g;\n', p.y{j}, yScale);
                        fprintf(fid, 'plot(ax, %s, yV, ''LineWidth'', %g, ''LineStyle'', ''%s'', ''Marker'', ''%s'', ''DisplayName'', ''%s'');\n', ...
                            p.x{j}, p.yWidths(j), cleanLS, cleanMarker, p.y{j});
                    end
                end
                
                parsedTitle = obj.fixLaTeXColors(p.title);
                if ~isempty(p.title), fprintf(fid, 'title(ax, ''%s'', ''Interpreter'', ''tex'');\n', parsedTitle); end
                if p.grid, fprintf(fid, 'grid(ax, ''on'');\n'); end
            end
            fclose(fid);
            obj.logMessage('SYSTEM', 'Standalone generation reproduction script written successfully.');
        end
