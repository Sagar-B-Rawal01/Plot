    methods (Access = private)
        function safeExecute(obj, functionHandle, startMsg)
            try
                if ~isempty(startMsg), obj.logMessage('ACTION', startMsg); end
                functionHandle();
            catch ME
                errMsg = sprintf('%s (Line %d)', ME.message, ME.stack(1).line);
                obj.logMessage('ERROR', errMsg); uialert(obj.UI.Fig, errMsg, 'Runtime Error');
            end
        end

        function logMessage(obj, level, msg)
            txt = sprintf('[%s] [%s] %s', datestr(now, 'HH:MM:SS'), level, msg);
            obj.UI.Console.Value{end+1} = txt; scroll(obj.UI.Console, 'bottom');
        end

        function out = safeStr(~, val)
            if ismissing(val) || (isnumeric(val) && isnan(val)), out = '';
            elseif ischar(val) || isstring(val), out = char(val);
            else, out = char(string(val)); end
        end

        function out = safeNum(~, val, def)
            if ismissing(val) || ~isnumeric(val) || isnan(val), out = def; else, out = val; end
        end

        function actionToggle3D(obj, ev)
            if ev.Value, obj.UI.pltZList.Enable = 'on'; else, obj.UI.pltZList.Enable = 'off'; end

        function actionLoadMat(obj)
            [file,path] = uigetfile('*.mat', 'Select MAT Files', 'MultiSelect','on');
            if isequal(file,0), return; end
            if ischar(file), file = {file}; end
            for k = 1:numel(file)
                fullFile = fullfile(path,file{k}); obj.LoadedFile = file{k}; obj.LoadedFiles{end+1} = fullFile;
                rawStruct = load(fullFile); obj.parseStructRecursive(rawStruct,'');
            end
            obj.actionRefreshTable(); obj.syncUILists();
            if ~isempty(obj.ExcelState(1).Sheet1), obj.actionRevalidateExcel(); end
        end

        function parseStructRecursive(obj,inStruct,prefix)
            fn = fieldnames(inStruct);
            for i = 1:length(fn)
                val = inStruct.(fn{i}); fullPath = fn{i}; if ~isempty(prefix), fullPath = [prefix '.' fn{i}]; end
                if isstruct(val), obj.parseStructRecursive(val,fullPath);
                else
                    if isnumeric(val) && size(val,1)>1 && size(val,2)>1
                        for c = 1:size(val,2)
                            safeName = obj.makeUniqueName(matlab.lang.makeValidName(sprintf('%s_col%d',fullPath,c)));
                            tokens = split(fullPath, '.');
                            label = char(sprintf('%s_%d', tokens{end}, c));
                            extractCmd = sprintf('%s(:,%d)', fullPath, c);
                            obj.DataVars.(safeName) = obj.createMetaStruct(val(:,c), safeName, label, extractCmd);
                        end
                    else
                        safeName = obj.makeUniqueName(matlab.lang.makeValidName(fullPath));
                        tokens = split(fullPath, '.');
                        label = char(tokens{end});
                        extractCmd = fullPath;
                        obj.DataVars.(safeName) = obj.createMetaStruct(val, safeName, label, extractCmd);
                    end
                end
            end
        end

        function meta = createMetaStruct(~, data, sourcePath, label, extractCmd)
            if nargin < 5, extractCmd = sourcePath; end
            meta = struct('data', data, 'sourcePath', sourcePath, 'label', char(label), ...
                'unit', '', 'scale', 1, 'offset', 0, 'transpose', false, ...
                'isDerived', false, 'expr', '', 'cache', [], 'delay', 0, 'extractCmd', extractCmd); 
        end

        function sig = fetchSignal(obj, key)
            meta = obj.DataVars.(key);
            if meta.isDerived
                if ismember(key, obj.EvalStack), error('Cyclic dependency: %s', key); end
                obj.EvalStack{end+1} = key;
                try sig = obj.evaluateMathSafely(meta.expr); obj.DataVars.(key).cache = sig; obj.EvalStack(end) = [];
                catch ME, obj.EvalStack = {}; rethrow(ME); end
                return;
            end
            if ~isempty(meta.cache), sig = meta.cache; return; end
            raw = meta.data; if istable(raw), raw = table2array(raw); end
            if isrow(raw), raw = raw(:); end
            if meta.transpose, raw = raw'; end
            raw = double(raw); if size(raw, 2) > 1, raw = raw(:,1); end
            sig = raw * meta.scale + meta.offset; sig(~isfinite(sig)) = NaN;
            
            if isfield(meta, 'delay') && meta.delay ~= 0
                sz = meta.delay;
                if sz > 0, sig = [nan(sz, 1); sig(1:end-sz)]; elseif sz < 0, sig = [sig(abs(sz)+1:end); nan(abs(sz), 1)]; end
            end
            obj.DataVars.(key).cache = sig;
        end

        function out = evaluateMathSafely(obj, expr)
            fn = fieldnames(obj.DataVars);
            for i = 1:length(fn)
                try eval(sprintf('%s = obj.fetchSignal(''%s'');', fn{i}, fn{i})); catch; end
            end
            out = eval(expr);
        end
        
        function [lS, marker] = parseLineStyle(~, rawStyle)
            lS = rawStyle; marker = 'none';
            if contains(lS, 'o'), marker = 'o'; lS = strrep(lS, 'o', ''); end
            if contains(lS, '.'), marker = '.'; lS = strrep(lS, '.', ''); end
            if contains(lS, 'x'), marker = 'x'; lS = strrep(lS, 'x', ''); end
            if isempty(lS), lS = 'none'; end
        end

        function actionCreateDerived(obj)
            name = char(strtrim(obj.UI.derName.Value)); expr = char(strtrim(obj.UI.derExpr.Value));
            safeName = matlab.lang.makeValidName(name); out = obj.evaluateMathSafely(expr);
            obj.DataVars.(safeName) = struct('data', out(:), 'sourcePath', safeName, 'label', name, ...
                'unit', char(obj.UI.derUnit.Value), 'scale', 1, 'offset', 0, 'transpose', false, ...
                'isDerived', true, 'expr', expr, 'cache', out(:), 'delay', 0, 'extractCmd', expr);
            obj.actionRefreshTable(); obj.syncUILists();
        end

        function actionCalculateStats(obj)
            keys = obj.getSelectedKeys();
            for i = 1:length(keys)
                k = keys{i}; validData = obj.fetchSignal(k); validData = validData(isfinite(validData));
                obj.logMessage('STATS', sprintf('Chan: %s | Max: %.2f | Min: %.2f | Mean: %.2f', k, max(validData), min(validData), mean(validData)));
            end
        end

        function actionRefreshTable(obj)
            k = lower(char(obj.UI.txtSearch.Value)); fn = fieldnames(obj.DataVars); d = cell(0, 9); rowIdx = 1; 
            for i = 1:length(fn)
                v = obj.DataVars.(fn{i});
                if isempty(k) || contains(lower(v.label), k) || contains(lower(fn{i}), k)
                    sz = size(v.data); if v.transpose, sz = fliplr(sz); end
                    if ~isfield(v, 'delay'), obj.DataVars.(fn{i}).delay = 0; v.delay = 0; end
                    d(rowIdx, :) = {fn{i}, v.label, v.unit, sprintf('%dx%d', sz(1), sz(2)), v.scale, v.offset, v.transpose, v.isDerived, v.delay};
                    rowIdx = rowIdx + 1;
                end
            end
            obj.UI.dataTable.Data = d;
        end

        function actionFilterList(obj, axisStr)
            if strcmp(axisStr, 'x')
                txt = lower(char(obj.UI.pltXSearch.Value)); list = obj.UI.pltXList;
                fn = fieldnames(obj.DataVars); matches = {}; lbls = {};
                for i = 1:length(fn)
                    lbl = char(obj.DataVars.(fn{i}).label);
                    if isempty(txt) || contains(lower(lbl), txt) || contains(lower(fn{i}), txt)
                        matches{end+1} = fn{i}; lbls{end+1} = lbl; 
                    end
                end
                list.Items = lbls; list.ItemsData = matches;
            elseif strcmp(axisStr, 'y')
                txt = lower(char(obj.UI.pltYSearch.Value)); fn = fieldnames(obj.DataVars);
                d = cell(0, 2); rowIdx = 1;
                for i = 1:length(fn)
                    lbl = char(obj.DataVars.(fn{i}).label);
                    if isempty(txt) || contains(lower(lbl), txt) || contains(lower(fn{i}), txt)
                        isSelected = any(strcmp(obj.SelectedYSignals, fn{i}));
                        d(rowIdx, :) = {isSelected, lbl};
                        rowIdx = rowIdx + 1;
                    end
                end
                obj.UI.pltYTable.Data = d;
            elseif strcmp(axisStr, 'z')
                txt = lower(char(obj.UI.pltZSearch.Value)); list = obj.UI.pltZList;
                fn = fieldnames(obj.DataVars); matches = {}; lbls = {};
                for i = 1:length(fn)
                    lbl = char(obj.DataVars.(fn{i}).label);
                    if isempty(txt) || contains(lower(lbl), txt) || contains(lower(fn{i}), txt)
                        matches{end+1} = fn{i}; lbls{end+1} = lbl; 
                    end
                end
                list.Items = lbls; list.ItemsData = matches;
            end
        end

            obj.UI.extExportList.Items = lbls; obj.UI.extExportList.ItemsData = fn;

        function syncManifest(obj)
            delete(obj.UI.plotTree.Children); 
            plotFigs = []; if ~isempty(obj.Plots), plotFigs = cellfun(@(x)x.f, obj.Plots); end
            evtFigs = []; if ~isempty(obj.Events), evtFigs = cellfun(@(x)x.fig, obj.Events); end
            allFigs = unique([plotFigs evtFigs]);

            for i = 1:length(allFigs)
                fNum = allFigs(i);
                figTitleStr = '';
                for p = 1:length(obj.Plots)
                    if obj.Plots{p}.f == fNum && ~isempty(obj.Plots{p}.fTitle)
                        figTitleStr = sprintf(': %s', obj.Plots{p}.fTitle);
                        break;
                    end
                end
                
                fNode = uitreenode(obj.UI.plotTree, 'Text', sprintf('Figure %d%s', fNum, figTitleStr), 'NodeData', struct('type', 'fig', 'f', fNum));
                
                for p = 1:length(obj.Plots)
                    if obj.Plots{p}.f == fNum
                        pName = sprintf('Pos %d: %s', obj.Plots{p}.p, strjoin(obj.Plots{p}.y, ','));
                        if obj.Plots{p}.is3D, pName = [pName ' (3D)']; end
                        uitreenode(fNode, 'Text', pName, 'NodeData', struct('type', 'plot', 'id', obj.Plots{p}.id));
                    end
                end
                expand(fNode);
            end
            obj.syncEventSegmentList();
        end

        function syncEventSegmentList(obj)
            items = {}; ids = {};
            for i = 1:length(obj.Events)
                e = obj.Events{i};
                for j = 1:length(e.onT)
                    items{end+1} = sprintf('[F%d] %s: Seg %d (%.2fs - %.2fs)', e.fig, e.name, j, e.onT(j), e.offT(j)); %#ok<AGROW>
                    ids{end+1} = struct('fig', e.fig, 't1', e.onT(j), 't2', e.offT(j)); %#ok<AGROW>
                end
            end
            
            if ~isempty(items)
                obj.UI.evSegList.Enable = 'on'; obj.UI.evSegList.Items = items; obj.UI.evSegList.ItemsData = ids;
            else
                obj.UI.evSegList.Enable = 'off'; obj.UI.evSegList.Items = {'No Segments'}; obj.UI.evSegList.ItemsData = {};
            end
        end

        function actionLoadExcel(obj)
            [file, path] = uigetfile({'*.xlsx';'*.xls'}, 'Select Excel Dashboard Template');
            if isequal(file,0), return; end
            fullPath = fullfile(path, file);

            try
                s1 = readcell(fullPath, 'Sheet', 1);
                s2 = readcell(fullPath, 'Sheet', 2);
            catch ME
                error('Ensure the Excel file has Sheet1 (Plots) and Sheet2 (Dictionary). %s', ME.message);
            end

            obj.ExcelState(1).Sheet1 = s1;
            obj.ExcelState(1).Sheet2 = s2;

        function actionRevalidateExcel(obj)
            if isempty(obj.ExcelState) || isempty(obj.ExcelState(1).Sheet1), return; end
            s1 = obj.ExcelState(1).Sheet1;
            s2 = obj.ExcelState(1).Sheet2;

            reqSigs = {};
            for i = 2:size(s1,1)
                x = obj.safeStr(s1{i,9}); if ~isempty(x), reqSigs{end+1} = x; end %#ok<AGROW>
                for c = 10:13
                    y = obj.safeStr(s1{i,c}); if ~isempty(y), reqSigs{end+1} = y; end %#ok<AGROW>
                end
                z = obj.safeStr(s1{i,14}); if ~isempty(z), reqSigs{end+1} = z; end %#ok<AGROW>
            end
            reqSigs = unique(reqSigs);

            mapDict = containers.Map();
            for i = 2:size(s2,1)
                k = obj.safeStr(s2{i,1}); cat = obj.safeStr(s2{i,2}); src = obj.safeStr(s2{i,3});
                if isempty(cat), cat = 'Uncategorized'; end
                if ~isempty(k)
                    mapDict(k) = cat;
                    if strcmpi(cat, 'Derived') && ~isfield(obj.DataVars, k) && ~isempty(src)
                        try
                            out = obj.evaluateMathSafely(src);
                            scale = obj.safeNum(s2{i,4}, 1); off = obj.safeNum(s2{i,5}, 0); delay = obj.safeNum(s2{i,6}, 0); unit = obj.safeStr(s2{i,7});
                            out = out * scale + off;
                            obj.DataVars.(k) = obj.createMetaStruct(out(:), k, k, src);
                            obj.DataVars.(k).unit = unit; obj.DataVars.(k).scale = 1; obj.DataVars.(k).offset = 0; obj.DataVars.(k).delay = delay; obj.DataVars.(k).isDerived = true; obj.DataVars.(k).expr = src;
                        catch
                        end
                    end
                end
            end
            
            obj.actionRefreshTable(); obj.syncUILists();

            catMap = containers.Map();
            for i = 1:length(reqSigs)
                sig = reqSigs{i};
                if isKey(mapDict, sig), cat = mapDict(sig); else, cat = 'Unmapped Variables'; end
                if isfield(obj.DataVars, sig), color = 'green'; else, color = 'red'; end
                htmlStr = sprintf('<html><font color="%s">%s</font></html>', color, sig);
                
                if isKey(catMap, cat), list = catMap(cat); list{end+1} = htmlStr; catMap(cat) = list;
                else, catMap(cat) = {htmlStr}; end
            end

            delete(obj.UI.valTree.Children);
            cats = catMap.keys();
            for i = 1:length(cats)
                cNode = uitreenode(obj.UI.valTree, 'Text', cats{i});
                sigs = catMap(cats{i});
                for j = 1:4:length(sigs)
                    chunk = sigs(j:min(j+3, length(sigs)));
                    uitreenode(cNode, 'Text', ['<html>' strjoin(chunk, ', ') '</html>']);
                end
                expand(cNode);
            end
        end

        function actionGenerateFromExcel(obj)
            if isempty(obj.ExcelState) || isempty(obj.ExcelState(1).Sheet1), error('Load Excel Template first.'); end
            s1 = obj.ExcelState(1).Sheet1;

            curFig = 1; curRow = 1; curCol = 1; curSg = ''; curSub = ''; curPos = 0;
            
            for i = 2:size(s1,1)
                figIn = obj.safeNum(s1{i,1}, NaN);
                if ~isnan(figIn)
                    if figIn ~= curFig, curPos = 0; end
                    curFig = figIn;
                end

                curRow = obj.safeNum(s1{i,2}, curRow); curCol = obj.safeNum(s1{i,3}, curCol);
                sg = obj.safeStr(s1{i,4}); if ~isempty(sg) || ~ismissing(s1{i,4}), curSg = sg; end
                su = obj.safeStr(s1{i,5}); if ~isempty(su) || ~ismissing(s1{i,5}), curSub = su; end

                maxP = curRow * curCol;
                posIn = obj.safeNum(s1{i,6}, NaN);
                if isnan(posIn)
                    curPos = curPos + 1;
                    if curPos > maxP, curPos = maxP; end
                else
                    curPos = posIn;
                    if curPos > maxP, curPos = maxP; end
                end

                is3D = false; raw3D = s1{i,8};
                if islogical(raw3D), is3D = raw3D; elseif isnumeric(raw3D) && raw3D==1, is3D=true; elseif ischar(raw3D) && strcmpi(raw3D,'true'), is3D=true; end

                xSig = obj.safeStr(s1{i,9}); if ~isfield(obj.DataVars, xSig), continue; end 
                ySigs = {};
                for c=10:13
                    y = obj.safeStr(s1{i,c}); if ~isempty(y) && isfield(obj.DataVars, y), ySigs{end+1} = y; end %#ok<AGROW>
                end
                if isempty(ySigs), continue; end 
                zSig = obj.safeStr(s1{i,14}); if is3D && ~isfield(obj.DataVars, zSig), continue; end 

                spTitle = obj.safeStr(s1{i,7});
                cCol = obj.safeStr(s1{i,15}); if isempty(cCol), cCol = 'auto'; end
                cLs = obj.safeStr(s1{i,16}); if isempty(cLs), cLs = '-'; end
                cLw = obj.safeNum(s1{i,17}, 1.5);

                id = char(javaMethod('randomUUID','java.util.UUID'));
                p = struct('id', id, 'f', curFig, 'r', curRow, 'c', curCol, 'p', curPos, ...
                    'color', cCol, 'lW', cLw, 'lS', cLs, ...
                    'grid', true, 'leg', true, 'title', spTitle, ...
                    'fTitle', curSg, 'fSub', curSub, ...
                    'link', 0, 'x', xSig, 'y', {ySigs}, 'is3D', is3D);
                if is3D, p.z = zSig; end

                obj.Plots{end+1} = p;
            end

            obj.syncManifest();
            obj.renderAll([]);
            obj.logMessage('EXCEL', 'Dashboard Generated Successfully from Validated Signals.');
        end

        % --- RENDER ENGINE ---
        function renderAll(obj, targetFig)
            allFigs = unique([cellfun(@(x)x.f, obj.Plots) cellfun(@(x)x.fig, obj.Events)]);
            if ~isempty(targetFig), allFigs = intersect(allFigs, targetFig); end

            for i = 1:length(allFigs), figure(allFigs(i)); clf(figure(allFigs(i))); end
            axMap = containers.Map('KeyType', 'double', 'ValueType', 'any'); 

            for i = 1:length(obj.Plots)
                p = obj.Plots{i}; if ~isempty(targetFig) && ~ismember(p.f, targetFig), continue; end
                ax = subplot(p.r, p.c, p.p, 'Parent', figure(p.f)); hold(ax, 'on');

                if p.link > 0, if isKey(axMap, p.link), axMap(p.link) = [axMap(p.link), ax]; else, axMap(p.link) = ax; end; end
                
                if ~isempty(p.fTitle) || ~isempty(p.fSub)
                    if ~isempty(p.fSub)
                        sgtitle(figure(p.f), {p.fTitle, p.fSub}, 'Interpreter', 'none');
                    else
                        sgtitle(figure(p.f), p.fTitle, 'Interpreter', 'none');
                    end
                end

                [cleanLS, cleanMarker] = obj.parseLineStyle(p.lS);
                
                if p.is3D
                    xV = obj.fetchSignal(p.x); yV = obj.fetchSignal(p.y{1}); zV = obj.fetchSignal(p.z);
                    len = min([length(xV), length(yV), length(zV)]);
                    
                    dName = obj.DataVars.(p.y{1}).label; if ~isempty(obj.DataVars.(p.y{1}).unit), dName = sprintf('%s (%s)', dName, obj.DataVars.(p.y{1}).unit); end
                    
                    args = {'LineWidth', p.lW, 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dName};
                    if ~strcmp(p.color, 'auto')
                        args = [args, 'Color', p.color]; 
                        if ~strcmp(cleanMarker, 'none'), args = [args, 'MarkerFaceColor', p.color]; end
                    end
                    
                    plot3(ax, xV(1:len), yV(1:len), zV(1:len), args{:});
                    xlabel(ax, obj.DataVars.(p.x).label, 'Interpreter', 'none'); ylabel(ax, obj.DataVars.(p.y{1}).label, 'Interpreter', 'none'); zlabel(ax, obj.DataVars.(p.z).label, 'Interpreter', 'none');
                    grid(ax, 'on'); view(ax, 3);
                else
                    xV = obj.fetchSignal(p.x); lbls = {};
                    for j = 1:length(p.y)
                        yV = obj.fetchSignal(p.y{j}); m = obj.DataVars.(p.y{j}); len = min(length(xV), length(yV));
                        
                        dName = char(m.label); if ~isempty(m.unit), dName = sprintf('%s (%s)', dName, m.unit); end
                        args = {'LineWidth', p.lW, 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dName};
                        if ~strcmp(p.color, 'auto')
                            args = [args, 'Color', p.color];
                            if ~strcmp(cleanMarker, 'none'), args = [args, 'MarkerFaceColor', p.color]; end
                        end
                        
                        plot(ax, xV(1:len), yV(1:len), args{:});
                        lbls{end+1} = dName; %#ok<AGROW>
                    end
                    xLbl = char(obj.DataVars.(p.x).label); if ~isempty(obj.DataVars.(p.x).unit), xLbl = sprintf('%s (%s)', xLbl, obj.DataVars.(p.x).unit); end
                    xlabel(ax, xLbl, 'Interpreter', 'none'); ylabel(ax, strjoin(lbls, ' / '), 'Interpreter', 'none');
                end

                if ~isempty(p.title), title(ax, p.title, 'Interpreter', 'none'); end
                if p.grid, grid(ax, 'on'); end
                if p.leg, legend(ax, 'Location', 'best', 'Interpreter', 'none'); end
            end

            linkIds = axMap.keys(); for i = 1:length(linkIds), if length(axMap(linkIds{i})) > 1, linkaxes(axMap(linkIds{i}), 'x'); end; end

            for i = 1:length(obj.Events)
                e = obj.Events{i}; if ~isempty(targetFig) && ~ismember(e.fig, targetFig), continue; end
                axesList = findall(figure(e.fig), 'Type', 'axes', '-not', 'Tag', 'legend');
                for a = 1:length(axesList)
                    ax = axesList(a); hold(ax, 'on');
                    for t = 1:length(e.onT)
                        xline(ax, e.onT(t), '-', 'Start', 'Color', e.c, 'LineWidth', 1.5, 'HandleVisibility', 'off');
                        xline(ax, e.offT(t), '--', 'Stop', 'Color', e.c, 'LineWidth', 1.5, 'HandleVisibility', 'off');
                    end
                end
            end
        end

        function actionAddEvent(obj)
            expr = char(strtrim(obj.UI.evLogic.Value)); 
            mask = obj.evaluateMathSafely(expr); mask = double(mask(:)); mask(~isfinite(mask)) = 0; mask(mask ~= 0) = 1;
            
            d = diff([0; mask; 0]); startIdx = find(d == 1); stopIdx = find(d == -1) - 1; 
            if isempty(startIdx), error('No logical event transitions found.'); end
            
            fn = fieldnames(obj.DataVars); labels = cell(size(fn));
            for i=1:length(fn), labels{i} = char(obj.DataVars.(fn{i}).label); end
            [indx, tf] = listdlg('PromptString', 'Select X-axis reference', 'SelectionMode', 'single', 'ListString', labels); 
            if ~tf, return; end
            
            t = obj.fetchSignal(fn{indx}); t = t(:);
            e = struct('id', char(javaMethod('randomUUID','java.util.UUID')), 'name', char(obj.UI.evName.Value), ...
                'onT', t(startIdx), 'offT', t(stopIdx), 'fig', obj.UI.evFig.Value, 'c', char(obj.UI.evColor.Value));
            obj.Events{end+1} = e; obj.syncManifest();
        end

            fprintf(fid, 'clearvars; clc;\n\n');

            fprintf(fid, '%% --- 1. Load Data ---\n');
            for f = 1:length(obj.LoadedFiles)
                fprintf(fid,'load(''%s'');\n',strrep(obj.LoadedFiles{f},'''', ''''''));
            end
            fprintf(fid, '\n');

            fprintf(fid, '%% --- 2. Format Raw Data ---\n');
            fn = fieldnames(obj.DataVars);
            for i = 1:length(fn)
                m = obj.DataVars.(fn{i});
                if ~m.isDerived
                    fprintf(fid, '%s = double(%s);\n', m.sourcePath, m.extractCmd);
                    if m.transpose, fprintf(fid, 'if size(%s, 1) < size(%s, 2), %s = %s''; end\n', m.sourcePath, m.sourcePath, m.sourcePath, m.sourcePath); end
                    fprintf(fid, 'if isrow(%s), %s = %s(:); end\n', m.sourcePath, m.sourcePath, m.sourcePath);

                    fprintf(fid, '%s = %s * %g + %g;\n', m.sourcePath, m.sourcePath, m.scale, m.offset);
                    if m.delay ~= 0
                        fprintf(fid, 'if %d > 0; %s = [nan(%d,1); %s(1:end-%d)]; else; %s = [%s(%d:end); nan(%d,1)]; end\n', m.delay, m.sourcePath, m.delay, m.sourcePath, m.delay, m.sourcePath, m.sourcePath, abs(m.delay)+1, abs(m.delay));
                    end

            end

                end

                    if ~isempty(p.fSub)
                        fprintf(fid, 'sgtitle(figure(%d), {''%s'', ''%s''}, ''Interpreter'', ''none'');\n', p.f, strrep(p.fTitle, '''', ''''''), strrep(p.fSub, '''', ''''''));
                    else
                        fprintf(fid, 'sgtitle(figure(%d), ''%s'', ''Interpreter'', ''none'');\n', p.f, strrep(p.fTitle, '''', ''''''));
                    end
                end

                [cleanLS, cleanMarker] = obj.parseLineStyle(p.lS);
                colorArg = ''; 
                if ~strcmp(p.color, 'auto')
                    colorArg = sprintf(', ''Color'', ''%s''', p.color); 
                    if ~strcmp(cleanMarker, 'none')
                        colorArg = sprintf('%s, ''MarkerFaceColor'', ''%s''', colorArg, p.color);
                    end
                end

                if p.is3D
                    fprintf(fid, 'plot3(ax, %s, %s, %s, ''LineWidth'', %g, ''LineStyle'', ''%s'', ''Marker'', ''%s'' %s);\n', ...
                        obj.DataVars.(p.x).sourcePath, obj.DataVars.(p.y{1}).sourcePath, obj.DataVars.(p.z).sourcePath, p.lW, cleanLS, cleanMarker, colorArg);
                    
                    fprintf(fid, 'xlabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(obj.DataVars.(p.x).label, '''', ''''''));
                    fprintf(fid, 'ylabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(obj.DataVars.(p.y{1}).label, '''', ''''''));
                    fprintf(fid, 'zlabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(obj.DataVars.(p.z).label, '''', ''''''));

                    for j = 1:length(p.y)
                        m = obj.DataVars.(p.y{j});
                        dName = char(m.label); if ~isempty(m.unit), dName = sprintf('%s (%s)', dName, m.unit); end
                        
                        fprintf(fid, 'plot(ax, %s, %s, ''LineWidth'', %g, ''LineStyle'', ''%s'', ''Marker'', ''%s'', ''DisplayName'', ''%s'' %s);\n', ...
                            obj.DataVars.(p.x).sourcePath, m.sourcePath, p.lW, cleanLS, cleanMarker, strrep(dName, '''', ''''''), colorArg);
                        lbls{end+1} = dName; %#ok<AGROW>
                    end
                    
                    xLbl = char(obj.DataVars.(p.x).label); if ~isempty(obj.DataVars.(p.x).unit), xLbl = sprintf('%s (%s)', xLbl, obj.DataVars.(p.x).unit); end
                    fprintf(fid, 'xlabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(xLbl, '''', ''''''));
                    fprintf(fid, 'ylabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(strjoin(lbls, ' / '), '''', ''''''));
                end
                
                if ~isempty(p.title), fprintf(fid, 'title(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(p.title, '''', '''''')); end
                if p.grid, fprintf(fid, 'grid(ax, ''on'');\n'); end
                if p.leg, fprintf(fid, 'legend(ax, ''Location'', ''best'', ''Interpreter'', ''none'');\n'); end
                
                fprintf(fid, '\n');
            end
            fprintf(fid, '%% --- 5. Apply Discrete Group Axis Linking ---\nlinkIds = axMap.keys();\nfor i = 1:length(linkIds);\n    axList = axMap(linkIds{i});\n    if length(axList) > 1, linkaxes(axList, ''x''); end\nend\n\n');
            fclose(fid);
        end
