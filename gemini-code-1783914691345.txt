        %% Core Logic & Regex Engines
        function safeExecute(obj, functionHandle, startMsg)
            try
                if ~isempty(startMsg), obj.logMessage('ACTION', startMsg); end
                functionHandle();
            catch ME
                errMsg = sprintf('%s (Line %d)', ME.message, ME.stack(1).line);
                obj.logMessage('ERROR', errMsg);
                uialert(obj.UI.Fig, errMsg, 'Runtime Error');
            end
        end
        
        function logMessage(obj, level, msg)
            txt = sprintf('[%s] [%s] %s', datestr(now, 'HH:MM:SS'), level, msg);
            obj.UI.Console.Value{end+1} = txt;
            scroll(obj.UI.Console, 'bottom');
        end

        function str = fixLaTeXColors(~, str)
            if isempty(str), return; end
            map = {'\color{r}', '\color{red}'; '\color{g}', '\color{green}'; 
                   '\color{b}', '\color{blue}'; '\color{c}', '\color{cyan}'; 
                   '\color{m}', '\color{magenta}'; '\color{y}', '\color{yellow}'; 
                   '\color{k}', '\color{black}'};
            for i = 1:size(map, 1)
                str = strrep(str, map{i, 1}, map{i, 2});
            end
        end

        function actionLoadMat(obj)
            [file,path] = uigetfile('*.mat', 'Select MAT Files', 'MultiSelect','on');
            if isequal(file,0), return; end
            if ischar(file), file = {file}; end
            
            prefixStr = inputdlg('Enter variable prefix (leave blank for none):', 'File Prefix', [1 40], {''});
            if isempty(prefixStr), pref = ''; else, pref = prefixStr{1}; end
            
            for k = 1:numel(file)
                fullFile = fullfile(path,file{k});
                obj.LoadedFile = file{k};
                obj.LoadedFiles{end+1} = fullFile;
                rawStruct = load(fullFile);
                obj.parseStructRecursive(rawStruct, '', pref);
            end
            obj.actionRefreshTable();
            obj.syncUILists();
        end
        
        function parseStructRecursive(obj, inStruct, prefixPath, namePrefix)
            fn = fieldnames(inStruct);
            for i = 1:length(fn)
                val = inStruct.(fn{i});
                fullPath = fn{i};
                if ~isempty(prefixPath), fullPath = [prefixPath '.' fn{i}]; end
                
                if isstruct(val)
                    obj.parseStructRecursive(val, fullPath, namePrefix);
                elseif isnumeric(val) || islogical(val)
                    safeName = [namePrefix matlab.lang.makeValidName(fullPath)];
                    safeName = obj.makeUniqueName(safeName);
                    
                    meta = struct('data', double(val(:)), 'label', safeName, 'unit', '', ...
                                  'offset', 0, 'transpose', false, 'isDerived', false, ...
                                  'expr', '', 'cache', [], 'delay', 0);
                    obj.DataVars.(safeName) = meta;
                end
            end
        end
        
        function safeName = makeUniqueName(obj, safeName)
            baseName = safeName; count = 1;
            while isfield(obj.DataVars, safeName)
                safeName = matlab.lang.makeValidName(sprintf('%s_%d', baseName, count));
                count = count + 1;
            end
        end
        
        function sig = fetchSignal(obj, key)
            meta = obj.DataVars.(key);
            if meta.isDerived
                sig = obj.evaluateMathSafely(meta.expr);
                obj.DataVars.(key).cache = sig;
                return;
            end
            if ~isempty(meta.cache), sig = meta.cache; return; end
            sig = meta.data + meta.offset;
            obj.DataVars.(key).cache = sig;
        end
        
        function out = evaluateMathSafely(obj, expr)
            vars = unique(regexp(expr, '[a-zA-Z]\w*', 'match'));
            for i = 1:length(vars)
                if isfield(obj.DataVars, vars{i})
                    eval(sprintf('%s = obj.fetchSignal(''%s'');', vars{i}, vars{i}));
                end
            end
            out = eval(expr);
        end
        
        %% Logic Library (Save & Persist Logic Rules)
        function actionSaveDerivedLogic(obj)
            name = char(strtrim(obj.UI.derName.Value));
            expr = char(strtrim(obj.UI.derExpr.Value));
            if isempty(name) || isempty(expr), return; end
            obj.SavedDerivedLogic.(matlab.lang.makeValidName(name)) = struct('name', name, 'expr', expr, 'unit', char(obj.UI.derUnit.Value));
            obj.syncLogicLists();
            obj.UI.derSaved.Value = name;
        end
        
        function actionLoadDerivedLogic(obj)
            key = char(obj.UI.derSaved.Value);
            if strcmp(key, '-- None --') || ~isfield(obj.SavedDerivedLogic, matlab.lang.makeValidName(key)), return; end
            s = obj.SavedDerivedLogic.(matlab.lang.makeValidName(key));
            obj.UI.derName.Value = s.name;
            obj.UI.derExpr.Value = s.expr;
            obj.UI.derUnit.Value = s.unit;
        end
        
        function actionSaveEventLogic(obj)
            name = char(strtrim(obj.UI.evName.Value));
            expr = char(strtrim(obj.UI.evLogic.Value));
            if isempty(name) || isempty(expr), return; end
            obj.SavedEventLogic.(matlab.lang.makeValidName(name)) = struct('name', name, 'expr', expr, 'c', char(obj.UI.evColor.Value));
            obj.syncLogicLists();
            obj.UI.evSaved.Value = name;
        end
        
        function actionLoadEventLogic(obj)
            key = char(obj.UI.evSaved.Value);
            if strcmp(key, '-- None --') || ~isfield(obj.SavedEventLogic, matlab.lang.makeValidName(key)), return; end
            s = obj.SavedEventLogic.(matlab.lang.makeValidName(key));
            obj.UI.evName.Value = s.name;
            obj.UI.evLogic.Value = s.expr;
            obj.UI.evColor.Value = s.c;
        end

        function syncLogicLists(obj)
            dKeys = fieldnames(obj.SavedDerivedLogic);
            if isempty(dKeys), obj.UI.derSaved.Items = {'-- None --'};
            else, obj.UI.derSaved.Items = cellfun(@(k) obj.SavedDerivedLogic.(k).name, dKeys, 'UniformOutput', false); end
            
            eKeys = fieldnames(obj.SavedEventLogic);
            if isempty(eKeys), obj.UI.evSaved.Items = {'-- None --'};
            else, obj.UI.evSaved.Items = cellfun(@(k) obj.SavedEventLogic.(k).name, eKeys, 'UniformOutput', false); end
        end
        
        function actionCreateDerived(obj)
            name = char(strtrim(obj.UI.derName.Value));
            expr = char(strtrim(obj.UI.derExpr.Value));
            safeName = matlab.lang.makeValidName(name);

        function actionValidateState(obj)
            reqVars = {};
            % Collect from Plots
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                if ischar(p.x), reqVars{end+1} = p.x; else, reqVars = [reqVars, p.x]; end
                reqVars = [reqVars, p.y];
                if p.is3D, reqVars{end+1} = p.z; end
                if p.is4D && ~isempty(p.sVar), reqVars{end+1} = p.sVar; end
            end
            
            % Collect from Derived signals
            fn = fieldnames(obj.DataVars);
            for i = 1:length(fn)
                if obj.DataVars.(fn{i}).isDerived
                    vars = unique(regexp(obj.DataVars.(fn{i}).expr, '[a-zA-Z]\w*', 'match'));
                    reqVars = [reqVars, vars];
                end
            end
            
            % Collect from Events
            for i = 1:length(obj.Events)
                vars = unique(regexp(obj.Events{i}.expr, '[a-zA-Z]\w*', 'match'));
                reqVars = [reqVars, vars];
            end
            
            reqVars = unique(reqVars);
            missingVars = {};
            for i = 1:length(reqVars)
                if ~isempty(reqVars{i}) && ~isfield(obj.DataVars, reqVars{i})
                    missingVars{end+1} = reqVars{i};
                end
            end
            
            if isempty(missingVars)
                uialert(obj.UI.Fig, 'All required variables for plots and events are present!', 'Validation Passed', 'Icon', 'success');
            else
                uialert(obj.UI.Fig, sprintf('Missing required variables:\n%s', strjoin(missingVars, '\n')), 'Validation Failed', 'Icon', 'error');
            end
        end

            [file, path] = uiputfile('*.mat', 'Save State'); if file == 0, return; end
            
            % Only save logic metadata, strip out large numeric arrays
            lightDataVars = struct();
            fn = fieldnames(obj.DataVars);
            for i = 1:length(fn)
                if obj.DataVars.(fn{i}).isDerived
                    lightDataVars.(fn{i}) = obj.DataVars.(fn{i});
                    lightDataVars.(fn{i}).data = [];
                    lightDataVars.(fn{i}).cache = [];
                end
            end
            
            % Strip heavy index tracking from events to allow mapping to new data
            lightEvents = obj.Events;
            for i=1:length(lightEvents)
                lightEvents{i}.onT = []; lightEvents{i}.offT = []; 
                lightEvents{i}.onIdx = []; lightEvents{i}.offIdx = [];
            end
            
            appState = struct('DerivedVars', lightDataVars, 'Plots', {obj.Plots}, 'Events', {lightEvents}, ...
                'SavedDerivedLogic', obj.SavedDerivedLogic, 'SavedEventLogic', obj.SavedEventLogic);

                    uitreenode(fNode, 'Text', pName, 'NodeData', struct('type', 'plot', 'id', pList{j}.id));

        %% Render Engine Execution
        function renderAll(obj, targetFig, isOverlay, targetGroups, targetPlots)
            if nargin < 3, isOverlay = false; end
            if nargin < 4, targetGroups = []; end
            if nargin < 5, targetPlots = {}; end
            
            allFigs = unique(cellfun(@(x)x.f, obj.Plots));
            if ~isempty(targetFig), allFigs = intersect(allFigs, targetFig); end
            
            if ~isOverlay
                for i = 1:length(allFigs), figure(allFigs(i)); clf(figure(allFigs(i))); end
            end
            
            axMap = containers.Map('KeyType', 'double', 'ValueType', 'any');
            distinctColors = lines(15);
            
            nodes = obj.UI.plotTree.SelectedNodes; activeGroups = [];
            if isOverlay && ~isempty(nodes)
                for i=1:length(nodes), if strcmp(nodes(i).NodeData.type, 'grp'), activeGroups(end+1) = nodes(i).NodeData.id; end; end
            end
            
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                
                if ~isempty(targetFig) && ~ismember(p.f, targetFig), continue; end
                if isOverlay && ~isempty(activeGroups) && ~ismember(p.group, activeGroups), continue; end
                if ~isOverlay && (~isempty(targetGroups) || ~isempty(targetPlots))
                    if ~ismember(p.group, targetGroups) && ~any(strcmp(p.id, targetPlots))
                        continue;
                    end
                end
                
                ax = subplot(p.r, p.c, p.p, 'Parent', figure(p.f)); hold(ax, 'on');
                if p.link > 0
                    if isKey(axMap, p.link), axMap(p.link) = [axMap(p.link), ax]; else, axMap(p.link) = ax; end
                end
                
                if ischar(p.x), p.x = repmat({p.x}, 1, length(p.y)); end
                if ~isfield(p, 'yLeg'), p.yLeg = true(1, length(p.y)); end
                
                existingTitle = ''; if isOverlay && ~isempty(ax.Title.String), existingTitle = ax.Title.String; end
                autoTitleChunks = {};
                firstValidX = '';
                
                if p.is4D
                    xSignal = p.x{1}; firstValidX = xSignal;
                    xV = obj.fetchSignal(xSignal);
                    yScale = 1; if isfield(p, 'yScales') && ~isempty(p.yScales), yScale = p.yScales(1); end
                    yV = obj.fetchSignal(p.y{1}) * yScale;
                    
                    cV = obj.fetchSignal(p.sVar);
                    len = min([length(xV), length(yV), length(cV)]);
                    [cleanLS, cleanMarker] = obj.parseLineStyle(p.lS);
                    
                    validC = cV(1:len);
                    
                    % Determine unique states for a discrete colormap
                    uStates = unique(validC(~isnan(validC)));
                    numStates = length(uStates);
                    if numStates == 0, numStates = 1; end
                    
                    cmap = lines(numStates);
                    colormap(ax, cmap);
                    
                    cMapped = zeros(size(validC));
                    for k = 1:numStates
                        cMapped(validC == uStates(k)) = k;
                    end
                    
                    caxis(ax, [1, numStates]);
                    cb = colorbar(ax);
                    cb.Ticks = 1:numStates;
                    
                    if isfield(p, 'tickLabels') && length(p.tickLabels) == numStates
                        cb.TickLabels = p.tickLabels;
                    else
                        cb.TickLabels = cellfun(@num2str, num2cell(uStates), 'UniformOutput', false);
                    end
                    
                    dispName = strrep(obj.DataVars.(p.y{1}).label, '_', '\_');
                    
                    if p.is3D
                        zV = obj.fetchSignal(p.z);
                        surface(ax, [xV(1:len) xV(1:len)], [yV(1:len) yV(1:len)], [zV(1:len) zV(1:len)], [cMapped(1:len)' cMapped(1:len)'], ...
                            'FaceColor', 'none', 'EdgeColor', 'flat', 'LineWidth', p.lW, 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dispName);
                    else
                        surface(ax, [xV(1:len) xV(1:len)], [yV(1:len) yV(1:len)], zeros(len, 2), [cMapped(1:len) cMapped(1:len)], ...
                            'FaceColor', 'none', 'EdgeColor', 'flat', 'LineWidth', p.lW, 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dispName);
                    end
                else
                    for j = 1:length(p.y)
                        xSignal = p.x{j}; 
                        xV = obj.fetchSignal(xSignal);
                        
                        yScale = 1; if isfield(p, 'yScales') && length(p.yScales) >= j, yScale = p.yScales(j); end
                        yStyle = p.lS; if isfield(p, 'yStyles') && length(p.yStyles) >= j, yStyle = p.yStyles{j}; end
                        
                        yV = obj.fetchSignal(p.y{j}) * yScale; len = min(length(xV), length(yV));
                        c = p.color;
                        
                        [cleanLS, cleanMarker] = obj.parseLineStyle(yStyle);
                        rawLabel = obj.DataVars.(p.y{j}).label;
                        
                        if p.yLeg(j)
                            if isempty(firstValidX), firstValidX = xSignal; end
                            dispName = strrep(rawLabel, '_', '\_');
                            visMode = 'on';
                        else
                            dispName = '';
                            visMode = 'off';
                        end
                        
                        args = {'LineWidth', p.lW, 'LineStyle', cleanLS, 'Marker', cleanMarker, 'DisplayName', dispName, 'HandleVisibility', visMode};
                        
                        if strcmp(c, 'auto')
                            cRGB = distinctColors(mod(length(findall(ax, 'Type', 'line')) + j - 1, 15) + 1, :);
                            args = [args, 'Color', cRGB];
                            if ~strcmp(cleanMarker, 'none'), args = [args, 'MarkerFaceColor', cRGB]; end
                            if p.yLeg(j)
                                autoTitleChunks{end+1} = sprintf('\\color[rgb]{%.3f,%.3f,%.3f}%s', cRGB(1), cRGB(2), cRGB(3), strrep(rawLabel, '_', '\_'));
                            end
                        else
                            args = [args, 'Color', c]; 
                            if ~strcmp(cleanMarker, 'none'), args = [args, 'MarkerFaceColor', c]; end
                            if p.yLeg(j)
                                mappedColor = obj.fixLaTeXColors(['\color{' c '}']);
                                autoTitleChunks{end+1} = sprintf('%s%s', mappedColor, strrep(rawLabel, '_', '\_'));
                            end
                        end
                        
                        if p.is3D
                            zV = obj.fetchSignal(p.z); plot3(ax, xV(1:len), yV(1:len), zV(1:len), args{:});
                        else
                            plot(ax, xV(1:len), yV(1:len), args{:});
                        end
                    end
                end
                
                if isempty(p.title) && ~isempty(autoTitleChunks)
                    plotTitleStr = strjoin(autoTitleChunks, ' \color{black}/ ');
                else
                    plotTitleStr = obj.fixLaTeXColors(p.title);
                end
                
                if ~isempty(existingTitle)
                    title(ax, [existingTitle ' \color{black}/ ' plotTitleStr], 'Interpreter', 'tex'); 
                else
                    if ~isempty(plotTitleStr), title(ax, plotTitleStr, 'Interpreter', 'tex'); end 
                end
                
                if ~isempty(p.xLabel)
                    xlabel(ax, p.xLabel, 'Interpreter', 'none'); 
                elseif ~isempty(firstValidX)
                    xlabel(ax, strrep(obj.DataVars.(firstValidX).label, '_', '\_'), 'Interpreter', 'none'); 
                end
                
                if ~isempty(p.yLabel), ylabel(ax, p.yLabel, 'Interpreter', 'none'); end
                if p.is3D && ~isempty(p.zLabel), zlabel(ax, p.zLabel, 'Interpreter', 'none'); end
                if p.grid, grid(ax, 'on'); end; if p.leg, legend(ax, 'Location', 'best'); end; if p.is3D, view(ax, 3); end
            end
            linkIds = axMap.keys(); for i = 1:length(linkIds), if length(axMap(linkIds{i})) > 1, linkaxes(axMap(linkIds{i}), 'x'); end; end
        end
        
        %% Advanced Event Engine & Interactive Data Tips
        function actionAddEvent(obj)
            expr = char(strtrim(obj.UI.evLogic.Value));
            mask = obj.evaluateMathSafely(expr);
            mask = double(mask(:)); mask(~isfinite(mask)) = 0; mask(mask ~= 0) = 1;
            
            d = diff([0; mask; 0]); startIdx = find(d == 1); stopIdx = find(d == -1) - 1;
            if isempty(startIdx), uialert(obj.UI.Fig, 'No logic transitions found.', 'Notice'); return; end
            
            fn = fieldnames(obj.DataVars);
            [indx, tf] = listdlg('PromptString', 'Select Time/X Reference Channel for logic evaluation:', 'SelectionMode', 'single', 'ListString', fn);
            if ~tf, return; end

        function renderEventsDirectly(obj, e)
            fNum = e.fig; if ~ishandle(fNum), return; end
            axList = findall(figure(fNum), 'Type', 'axes', '-not', 'Tag', 'legend');
            if e.pos > length(axList), return; end
            ax = axList(end - e.pos + 1); hold(ax, 'on');
            
            for t = 1:length(e.onT)
                xline(ax, e.onT(t), '-', 'Color', e.c, 'LineWidth', 1.5, 'HandleVisibility', 'off');
                xline(ax, e.offT(t), '--', 'Color', e.c, 'LineWidth', 1.5, 'HandleVisibility', 'off');
            end
        end
        
        function actionAddCustomDatatip(obj)
            figNum = obj.UI.evFig.Value; posNum = obj.UI.evPos.Value;
            if ~ishandle(figNum), uialert(obj.UI.Fig, 'Target Figure is not open.', 'Error'); return; end
            
            axList = findall(figure(figNum), 'Type', 'axes', '-not', 'Tag', 'legend');
            if isempty(axList) || posNum > length(axList)
                uialert(obj.UI.Fig, 'Target subplots position not found.', 'Error'); return;
            end
            ax = axList(end - posNum + 1);
            
            curves = findobj(ax, 'Type', 'line', '-or', 'Type', 'surface');
            if isempty(curves)
                uialert(obj.UI.Fig, 'No plotted line curves found on target subplot.', 'Error'); return;
            end
            
            names = cell(length(curves), 1);
            for i=1:length(curves)
                if ~isempty(curves(i).DisplayName), names{i} = curves(i).DisplayName;
                else, names{i} = sprintf('Curve Object (%d)', i); end
            end
            [cIdx, tf] = listdlg('PromptString', 'Select target line object to attach tip:', 'SelectionMode', 'single', 'ListString', names);
            if ~tf, return; end
            targetCurve = curves(cIdx);
            
            if isempty(obj.Events), uialert(obj.UI.Fig, 'No events available. Process an event first.', 'Error'); return; end
            evNames = cellfun(@(x) x.name, obj.Events, 'UniformOutput', false);
            [eIdx, tf2] = listdlg('PromptString', 'Select Event to mark:', 'SelectionMode', 'single', 'ListString', evNames);
            if ~tf2, return; end
            e = obj.Events{eIdx};
            
            xl = ax.XLabel.String; yl = targetCurve.DisplayName;
            if isempty(xl), xl = 'XData'; end; if isempty(yl), yl = 'YData'; end
            
            xData = targetCurve.XData; yData = targetCurve.YData; zData = [];
            if isprop(targetCurve, 'ZData') && ~isempty(targetCurve.ZData) && any(targetCurve.ZData(:) ~= 0)
                zData = targetCurve.ZData;
            end
            
            N = numel(xData);
            evtText = repmat({''}, N, 1);
            tipIndices = [];
            
            for t = 1:length(e.onIdx)
                idxOn = e.onIdx(t);
                if idxOn <= N
                    evtText{idxOn} = sprintf('%s [ON]', e.name);
                    tipIndices(end+1) = idxOn;
                end
                
                idxOff = e.offIdx(t);
                if idxOff <= N
                    evtText{idxOff} = sprintf('%s [OFF]', e.name);
                    tipIndices(end+1) = idxOff;
                end
            end
            
            dtt = targetCurve.DataTipTemplate;
            dtt.DataTipRows(1).Label = xl;
            dtt.DataTipRows(2).Label = yl;
            if ~isempty(zData) && numel(dtt.DataTipRows) >= 3
                zl = ax.ZLabel.String; if isempty(zl), zl = 'ZData'; end
                dtt.DataTipRows(3).Label = zl;
                while numel(dtt.DataTipRows) > 3, dtt.DataTipRows(end) = []; end
            else
                while numel(dtt.DataTipRows) > 2, dtt.DataTipRows(end) = []; end
            end
            
            dtt.DataTipRows(end+1) = dataTipTextRow('Event:', evtText);
            
            for i = 1:length(tipIndices)
                idx = tipIndices(i);
                if ~isempty(zData)
                    datatip(targetCurve, xData(idx), yData(idx), zData(idx));
                else
                    datatip(targetCurve, xData(idx), yData(idx));
                end
            end
            
            obj.logMessage('SYSTEM', sprintf('Automated datatips for %s attached successfully strictly by index.', e.name));
        end
        
        %% Segments & Isolation Infrastructure
        function syncEventList(obj)
            items = {}; ids = {};
            for i = 1:length(obj.Events)
                e = obj.Events{i};
                for j = 1:length(e.onT)
                    items{end+1} = sprintf('[F%d:P%d] %s: Seg %d (%.1fs - %.1fs)', e.fig, e.pos, e.name, j, e.onT(j), e.offT(j));
                    ids{end+1} = struct('fig', e.fig, 't1', e.onT(j), 't2', e.offT(j));
                end
            end
            if isempty(items)
                obj.UI.evSegList.Enable = 'off'; obj.UI.evSegList.Items = {'No Segments'}; obj.UI.evSegList.ItemsData = {};
            else
                obj.UI.evSegList.Enable = 'on'; obj.UI.evSegList.Items = items; obj.UI.evSegList.ItemsData = ids;
            end
        end

        function actionZoomSegment(obj)
            seg = obj.UI.evSegList.Value; if isempty(seg) || ~isstruct(seg), return; end
            pad = obj.UI.evZoomPad.Value;
            if ishandle(seg.fig)
                axList = findall(figure(seg.fig), 'Type', 'axes', '-not', 'Tag', 'legend');
                for a = 1:length(axList), xlim(axList(a), [seg.t1 - pad, seg.t2 + pad]); end
            end
        end
        
        function actionPlotSegment(obj)
            seg = obj.UI.evSegList.Value; if isempty(seg) || ~isstruct(seg), return; end
            activePlots = {};
            for p=1:length(obj.Plots), if obj.Plots{p}.f == seg.fig, activePlots{end+1} = obj.Plots{p}; end; end
            if isempty(activePlots), return; end
            
            f = figure('Name', 'Segment Extracted Plot'); ax = axes(f); hold(ax, 'on');
            for p=1:length(activePlots)
                
                if ischar(activePlots{p}.x), activePlots{p}.x = repmat({activePlots{p}.x}, 1, length(activePlots{p}.y)); end
                
                for y=1:length(activePlots{p}.y)
                    xV = obj.fetchSignal(activePlots{p}.x{y});
                    yScale = 1; if isfield(activePlots{p}, 'yScales') && length(activePlots{p}.yScales) >= y, yScale = activePlots{p}.yScales(y); end
                    yV = obj.fetchSignal(activePlots{p}.y{y}) * yScale;
                    idx = (xV >= seg.t1) & (xV <= seg.t2);
                    plot(ax, xV(idx), yV(idx), 'LineWidth', 1.5, 'DisplayName', activePlots{p}.y{y});
                end
            end
            legend(ax, 'show', 'Interpreter', 'none'); title(ax, 'Isolated Event Segment Duration'); grid(ax, 'on');
        end

        function actionYTableEdit(obj, ev)
            r = ev.Indices(1); c = ev.Indices(2);
            lbl = obj.UI.pltYTable.Data{r, 2};
            fn = fieldnames(obj.DataVars); key = '';
            for i = 1:length(fn), if strcmp(char(obj.DataVars.(fn{i}).label), lbl), key = fn{i}; break; end; end
            
            idx = find(strcmp(obj.SelectedYSignals, key));
            
            if c == 1
                if ev.NewData
                    if isempty(idx)
                        obj.SelectedYSignals{end+1} = key;
                        defX = char(obj.UI.pltXList.Value);
                        obj.SelectedXSignals{end+1} = defX;
                        obj.SelectedYScales(end+1) = obj.UI.pltYTable.Data{r, 4};
                        obj.SelectedYStyles{end+1} = obj.UI.pltYTable.Data{r, 5};
                        obj.SelectedYLeg(end+1) = obj.UI.pltYTable.Data{r, 6};
                        
                        if isfield(obj.DataVars, defX)
                            obj.UI.pltYTable.Data{r, 3} = char(obj.DataVars.(defX).label);
                        end
                    end
                else
                    if ~isempty(idx)
                        obj.SelectedYSignals(idx) = [];
                        obj.SelectedXSignals(idx) = [];
                        obj.SelectedYScales(idx) = [];
                        obj.SelectedYStyles(idx) = [];
                        obj.SelectedYLeg(idx) = [];
                    end
                end
            elseif c == 3 && ~isempty(idx)
                xLbl = ev.NewData;
                xKey = '';
                for i = 1:length(fn), if strcmp(char(obj.DataVars.(fn{i}).label), xLbl), xKey = fn{i}; break; end; end
                obj.SelectedXSignals{idx} = xKey;
            elseif c == 4 && ~isempty(idx)
                obj.SelectedYScales(idx) = ev.NewData;
            elseif c == 5 && ~isempty(idx)
                obj.SelectedYStyles{idx} = ev.NewData;
            elseif c == 6 && ~isempty(idx)
                obj.SelectedYLeg(idx) = ev.NewData;
            end
        end

            fn = fieldnames(obj.DataVars); 
            
            lbls = cell(1, length(fn));

            obj.UI.pltXList.Items = lbls; obj.UI.pltXList.ItemsData = fn;
            obj.UI.pltZList.Items = lbls; obj.UI.pltZList.ItemsData = fn;
            obj.UI.pltCList.Items = lbls; obj.UI.pltCList.ItemsData = fn;
            obj.UI.pltSList.Items = lbls; obj.UI.pltSList.ItemsData = fn;
            
            if isempty(lbls)
                obj.UI.pltYTable.ColumnFormat{3} = {'-- None --'};
            else
                obj.UI.pltYTable.ColumnFormat{3} = lbls(:)'; 
            end
            
            d = cell(length(fn), 6);
            for i=1:length(fn)
                idx = find(strcmp(obj.SelectedYSignals, fn{i}), 1);
                if ~isempty(idx)
                    xKey = obj.SelectedXSignals{idx};
                    if isempty(xKey) || ~isfield(obj.DataVars, xKey)
                        xLabel = '-- None --';
                    else
                        xLabel = char(obj.DataVars.(xKey).label);
                    end
                    d(i,:) = {true, lbls{i}, xLabel, obj.SelectedYScales(idx), obj.SelectedYStyles{idx}, obj.SelectedYLeg(idx)};
                else
                    defaultXLabel = ''; 
                    if ~isempty(obj.UI.pltXList.Value)
                        val = char(obj.UI.pltXList.Value);
                        if isfield(obj.DataVars, val)
                            defaultXLabel = char(obj.DataVars.(val).label); 
                        end
                    end
                    d(i,:) = {false, lbls{i}, defaultXLabel, 1, '-', true};
                end
            end
            obj.UI.pltYTable.Data = d;
        end

            txtField = sprintf('plt%sSearch', upper(axisStr)); txt = lower(char(obj.UI.(txtField).Value));

        function actionExportScript(obj)
            [file, path] = uiputfile('*.m', 'Save Render Script', 'telemetry_render_pro.m');
            if file == 0, return; end
            fid = fopen(fullfile(path, file), 'w');
            
            fprintf(fid, '%% STANDALONE PROCEDURAL REPRODUCTION CODE\nclearvars; clc;\n\n%% 1. Loading Vectors\n');
            for f = 1:length(obj.LoadedFiles), fprintf(fid,'load(''%s'');\n', strrep(obj.LoadedFiles{f},'''','''''')); end
            
            fprintf(fid, '\n%% 2. Formula Evaluation\n');
            dKeys = fieldnames(obj.SavedDerivedLogic);
            for i=1:length(dKeys)
                s = obj.SavedDerivedLogic.(dKeys{i}); fprintf(fid, '%s = %s;\n', s.name, s.expr);
            end
            
            fprintf(fid, '\n%% 3. Plot Manifest Loop Execution\n');
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                if ischar(p.x), p.x = repmat({p.x}, 1, length(p.y)); end
                
                fprintf(fid, '\nfigure(%d); ax = subplot(%d, %d, %d); hold(ax, ''on'');\n', p.f, p.r, p.c, p.p);
                
                if p.is4D
                    fprintf(fid, 'cV = %s;\n', p.cVar);
                    fprintf(fid, 'validC = cV(~isnan(cV));\n');
                    fprintf(fid, 'uStates = unique(validC);\n');
                    fprintf(fid, 'numStates = length(uStates); if numStates == 0, numStates = 1; end\n');
                    
                    fprintf(fid, 'colormap(ax, lines(numStates));\n');
                    fprintf(fid, 'cMapped = zeros(size(validC));\n');
                    fprintf(fid, 'for k = 1:numStates, cMapped(validC == uStates(k)) = k; end\n');
                    fprintf(fid, 'caxis(ax, [1, numStates]);\n');
                    
                    if isfield(p, 'sVar') && ~isempty(p.sVar) && ~isempty(p.tickVals)
                        fprintf(fid, 'cb = colorbar(ax);\n');
                        fprintf(fid, 'cb.Ticks = 1:numStates;\n');
                        
                        labelStr = '{';
                        for lblIdx = 1:length(p.tickLabels)
                            if lblIdx > 1, labelStr = [labelStr ', ']; end
                            labelStr = [labelStr '''' p.tickLabels{lblIdx} ''''];
                        end
                        labelStr = [labelStr '}'];
                        fprintf(fid, 'cb.TickLabels = %s;\n', labelStr);
                    end
                    
                    fprintf(fid, 'surface(ax, [%s, %s], [%s, %s], zeros(length(%s),2), [cMapped, cMapped], ''FaceColor'',''none'', ''EdgeColor'',''flat'', ''LineWidth'', %g);\n', ...
                        p.x{1}, p.x{1}, p.y{1}, p.y{1}, p.x{1}, p.lW);
                else
                    for j = 1:length(p.y)
                        yScale = 1; if isfield(p, 'yScales') && length(p.yScales) >= j, yScale = p.yScales(j); end
                        yStyle = p.lS; if isfield(p, 'yStyles') && length(p.yStyles) >= j, yStyle = p.yStyles{j}; end
                        [cleanLS, cleanMarker] = obj.parseLineStyle(yStyle);
                        
                        fprintf(fid, 'yV = %s * %g;\n', p.y{j}, yScale);
                        fprintf(fid, 'plot(ax, %s, yV, ''LineWidth'', %g, ''LineStyle'', ''%s'', ''Marker'', ''%s'', ''DisplayName'', ''%s'');\n', ...
                            p.x{j}, p.lW, cleanLS, cleanMarker, p.y{j});
                    end
                end
                
                parsedTitle = obj.fixLaTeXColors(p.title);
                if ~isempty(p.title), fprintf(fid, 'title(ax, ''%s'', ''Interpreter'', ''tex'');\n', parsedTitle); end
                if p.grid, fprintf(fid, 'grid(ax, ''on'');\n'); end
            end
            fclose(fid);
            obj.logMessage('SYSTEM', 'Standalone generation reproduction script written successfully.');
        end
