%% --- CORE ARRAYS DATA EXTRACTION ---
    function data = getProcessedData(varName, explicitCol)

    data = [];

    if ~isfield(dataVars, varName)
        return;
    end

    meta = dataVars.(varName);
    raw  = meta.data;

    if isempty(raw)
        return;
    end

    if istable(raw)
        raw = table2array(raw);
    end

    if meta.transpose
        raw = raw';
    end

    raw = double(raw);

    % Force vector orientation handling
    if isvector(raw)
        data = raw(:);

    else

        ncol = size(raw,2);

        if nargin >= 2 && ~isempty(explicitCol)

            explicitCol = min(explicitCol, ncol);
            data = raw(:, explicitCol);

        else

            % Default behavior
            if ncol >= 2
                data = raw(:,2);
            else
                data = raw(:,1);
            end
        end
    end

    data = data * meta.scale + meta.offset;

    data = data(:);

    % Remove inf/nan explosions
    bad = ~isfinite(data);
    data(bad) = NaN;
end

    function tData = getTimelineData(varName, explicitCol)

    tData = [];

    if ~isfield(dataVars, varName)
        return;
    end

    raw = dataVars.(varName).data;

    if isempty(raw)
        return;
    end

    if dataVars.(varName).transpose
        raw = raw';
    end

    raw = double(raw);

    if isvector(raw)

        tData = (1:length(raw))';

    else

        ncol = size(raw,2);

        if nargin >= 2 && ~isempty(explicitCol)

            explicitCol = min(explicitCol, ncol);
            tData = raw(:, explicitCol);

        else

            tData = raw(:,1);
        end
    end

    tData = tData(:);

    bad = ~isfinite(tData);
    tData(bad) = NaN;
end

%% --- FEATURE: STATE SAVE AND LOAD MECHANICS (SAVING INPUT GUI STATE) ---
    function saveAppState(~,~)
        [file, path] = uiputfile('*.mat', 'Save Layout Configuration Layout State', 'saved_layout_config.mat');
        if isequal(file,0), return; end
        % Pack app environment metrics safely into structured variables mapping parameters
        configState = struct();
        configState.plots = plots;
        configState.appEvents = appEvents;
        configState.layoutMapping = layoutMapping;
        configState.tableData = table.Data;
        configState.loadedFileName = loadedFileName;
        
        save(fullfile(path, file), 'configState');
        uialert(fig, 'App GUI layout configuration successfully stored!', 'State Stored');
    end

    function loadAppState(~,~)
        [file, path] = uigetfile('*.mat', 'Select Layout Configuration Layout State');
        if isequal(file,0), return; end
        tmp = load(fullfile(path, file));
        if ~isfield(tmp, 'configState')
            uialert(fig, 'Invalid state format selected.', 'Data Incompatible'); return;
        end
        
        % Unpack variables safely to restore GUI inputs layout
        plots = tmp.configState.plots;
        appEvents = tmp.configState.appEvents;
        layoutMapping = tmp.configState.layoutMapping;
        loadedFileName = tmp.configState.loadedFileName;
        
        % Sync listboxes display items
        plotList.Items = {};
        for m = 1:length(layoutMapping)
            mType = layoutMapping(m).type; mIdx = layoutMapping(m).idx;
            if strcmp(mType, 'plot')
                p = plots{mIdx};
                if p.isCustom
                    plotList.Items{end+1} = sprintf('[Math] F%d(S%d) | %s', p.figNum, p.pos, p.y{1});
                else
                    plotList.Items{end+1} = sprintf('[Plot] F%d(S%d) | %s', p.figNum, p.pos, strjoin(p.y,','));
                end
            elseif strcmp(mType, 'event')
                plotList.Items{end+1} = sprintf('[Region] Fig%d: %s', appEvents{mIdx}.figNum, appEvents{mIdx}.name);
            end
        end
        uialert(fig, 'Layout matrix state successfully restored. Click Preview to verify waves.', 'State Restored');
    end

    function addPlot(~,~)
        if isempty(xList.Value) || isempty(yList.Value), return; end
        p = struct('x',xList.Value,'y',{yList.Value},'type',plotType.Value,'row',subRow.Value,'col',subCol.Value,...
                   'pos',subIdx.Value,'figNum',fNum.Value,'color',colorDrop.Value,'xCol',xColField.Value,'yCol',yColField.Value,...
                   'linkGroup',linkGroupField.Value,'isCustom',false);
        plots{end+1} = p; 
        layoutMapping(end+1).type = 'plot'; layoutMapping(end).idx = length(plots);
        
        items = plotList.Items; 
        items{end+1} = sprintf('[Plot] F%d(S%d) Grp:%d | %s', p.figNum, p.pos, p.linkGroup, strjoin(p.y,','));
        plotList.Items = items;
    end

    function addCustomPlot(~,~)
        exprStr = customExprField.Value;
        if isempty(exprStr) || isempty(xList.Value), return; end
        p = struct('x',xList.Value,'y',{{exprStr}},'customName',customNameField.Value,'type',plotType.Value,'row',subRow.Value,'col',subCol.Value,...
                   'pos',subIdx.Value,'figNum',fNum.Value,'color',colorDrop.Value,'xCol',xColField.Value,...

        items = plotList.Items; 
        items{end+1} = sprintf('[Math] F%d(S%d) Grp:%d | %s', p.figNum, p.pos, p.linkGroup, exprStr);
        plotList.Items = items;
        previewPlot();
    end

    function addArrayLogicEvent(~,~)
        exprStr = eEquation.Value;
        if isempty(exprStr), return; end
        
        ws = struct(); fn = fieldnames(dataVars);
        for idx = 1:length(fn), ws.(fn{idx}) = getProcessedData(fn{idx}); end
        
        try
            logicalMask = evaluateLogicalExpression(exprStr, ws); logicalMask = double(logicalMask(:)); 
        catch ME
            uialert(fig, sprintf('Syntax error:\n%s', ME.message), 'Expression Error'); return;
        end
        
        tData = [];
        if ~isempty(xList.Value), tData = getTimelineData(xList.Value, xColField.Value); end
        if isempty(tData), tData = (1:length(logicalMask))'; end
        
        minLen = min(length(tData), length(logicalMask)); t = tData(1:minLen); mask = logicalMask(1:minLen);
        
        deltaChanges = [0; diff(mask)];
        onIndices  = find(deltaChanges == 1); offIndices = find(deltaChanges == -1);
        
        if mask(1) == 1, onIndices = [1; onIndices]; end
        if mask(end) == 1 && (isempty(offIndices) || offIndices(end) < onIndices(end)), offIndices = [offIndices; minLen]; end
        
        lenPairs = min(length(onIndices), length(offIndices));
        onTimes  = t(onIndices(1:lenPairs)); offTimes = t(offIndices(1:lenPairs));
        
        if isempty(onTimes), uialert(fig, 'No active thresholds seen.', 'No Regions'); return; end
        
        ev = struct('name',eName.Value,'expression',exprStr,'onTimes',onTimes,'offTimes',offTimes,...
                    'figNum',eFig.Value,'subplotMode',eSubplotMode.Value,'subplotIdx',eSubplotIdx.Value,...
                    'color',eColor.Value,'opacity',eOpacity.Value,...
                    'isolate',eIsolateCheck.Value,'buffer',eBuffer.Value);
        appEvents{end+1} = ev;
        layoutMapping(end+1).type = 'event'; layoutMapping(end).idx = length(appEvents);
        
        items = plotList.Items; 
        items{end+1} = sprintf('[Region] Fig%d: %s', ev.figNum, ev.name);
        plotList.Items = items; previewPlot();
    end

   function deleteSelected(~,~)

    val = plotList.Value;

    if isempty(val)
        return;
    end

    if iscell(val)
        val = val{1};
    end

    items = plotList.Items;

    idxToDelete = find(strcmp(items, val), 1);

    if isempty(idxToDelete)
        return;
    end

    itemType = layoutMapping(idxToDelete).type;
    itemIdx  = layoutMapping(idxToDelete).idx;

    if strcmp(itemType, 'plot')

        plots(itemIdx) = [];

        for m = 1:length(layoutMapping)

            if strcmp(layoutMapping(m).type,'plot') && ...
               layoutMapping(m).idx > itemIdx

                layoutMapping(m).idx = ...
                    layoutMapping(m).idx - 1;

            end
        end

    elseif strcmp(itemType, 'event')

        appEvents(itemIdx) = [];

        for m = 1:length(layoutMapping)

            if strcmp(layoutMapping(m).type,'event') && ...
               layoutMapping(m).idx > itemIdx

                layoutMapping(m).idx = ...
                    layoutMapping(m).idx - 1;

            end
        end
    end

    layoutMapping(idxToDelete) = [];

    items(idxToDelete) = [];

    plotList.Items = items;

    if isempty(items)
        plotList.Value = {};
    else
        plotList.Value = items{1};
    end

    previewPlot();

end

%% --- RENDER ENGINE FOR APP PREVIEW ---
    function previewPlot(~,~)
        allFigs = [];
        for idx = 1:length(plots)
            allFigs(end+1) = plots{idx}.figNum; %#ok<AGROW>
        end
        for idx = 1:length(appEvents)
            allFigs(end+1) = appEvents{idx}.figNum; %#ok<AGROW>
            if appEvents{idx}.isolate
                for s = 1:length(appEvents{idx}.onTimes)
                    allFigs(end+1) = 200 + (idx * 10) + s; %#ok<AGROW>
                end
            end
        end
        uniqueFigs = unique(allFigs(allFigs > 0));
        for f = uniqueFigs, if ishandle(f), clf(f); figure(f); else, figure(f); end; end

        linkMap = containers.Map('KeyType','double','ValueType','any');
        axMap = containers.Map;
        % Plot Waveforms Loop
        for k = 1:length(plots)
            p = plots{k};
            axKey = sprintf('F%d_R%d_C%d_P%d', ...
                p.figNum, p.row, p.col, p.pos);

            if isKey(axMap, axKey)

                ax = axMap(axKey);

            else

                figHandle = figure(p.figNum);

                ax = subplot(p.row, p.col, p.pos, ...
                    'Parent', figHandle);

                hold(ax, 'on');

                grid(ax, 'on');

                set(ax, 'UserData', p.pos);

                axMap(axKey) = ax;

            end
            xVal = getTimelineData(p.x, p.xCol);

            if p.isCustom
                ws = struct(); fn = fieldnames(dataVars);
                for idx = 1:length(fn), ws.(fn{idx}) = getProcessedData(fn{idx}); end
                try
                    yV = getProcessedData(yN, p.yCol);

                    xVal = xVal(:);
                    yV = yV(:);

                    len = min(length(xVal), length(yV));
                    disp(yN)
                    disp(size(xVal))
                    disp(size(yV))
                    disp([min(xVal) max(xVal)])
                    disp([min(yV) max(yV)])

                    if len <= 1
                        continue;
                    end
                    args = {'LineWidth', 1.5, 'DisplayName', p.customName};
                    if ~strcmp(p.color,'auto'), args = [args, 'Color', p.color]; end
                    hold(ax,'on');

                    plot(ax, xVal(1:len), yV(1:len), ...
                        'LineWidth',1.5, ...
                        'DisplayName', p.y{1});
                catch
                end
            else
                for j = 1:length(p.y)
                    yN = p.y{j}; yV = getProcessedData(yN, p.yCol); len = min(length(xVal), length(yV));
                    args = {'LineWidth', dataVars.(yN).width, 'DisplayName', dataVars.(yN).label};
                    if ~strcmp(p.color,'auto'), args = [args, 'Color', p.color]; end
                    feval(p.type, ax, xVal(1:len), yV(1:len), dataVars.(yN).style, args{:});
                end
            end
            if gridCheck.Value, grid(ax,'on'); end
            if legendCheck.Value, legend(ax,'show'); end

            % Build ylabel using displayed legend names
            legendNames = {};

            if p.isCustom

                legendNames{1} = p.customName;

            else

                for yy = 1:length(p.y)

                    yName = p.y{yy};

                    legendNames{end+1} = dataVars.(yName).label;

                end

            end
            legendNames=[pl,legendNames];

            ylabel(ax, strjoin(legendNames, ' / '));

            pl=legendNames;
            if p.linkGroup > 0
                if isKey(linkMap, p.linkGroup), linkMap(p.linkGroup) = [linkMap(p.linkGroup), ax]; else, linkMap(p.linkGroup) = ax; end
            end
        end

        % Inject Shading Regions & Dynamic Figure-Per-Event Set Zoom Engine
        for k = 1:length(appEvents)
            ev = appEvents{k}; targetFig = figure(ev.figNum);
            axList = findobj(targetFig, 'type', 'axes');
            if isempty(axList), axList = axes('Parent', targetFig); hold(axList, 'on'); end
            
            for a = 1:length(axList)
                currentAx = axList(a);
                if strcmp(ev.subplotMode, 'Specific Subplot Index')
                    subUserData = get(currentAx, 'UserData');
                    if ~isempty(subUserData) && subUserData ~= ev.subplotIdx, continue; end
                end
                for t = 1:length(ev.onTimes)
                    xregion(currentAx, ev.onTimes(t), ev.offTimes(t), 'FaceColor', ev.color, 'FaceAlpha', ev.opacity, 'HandleVisibility', 'off');
                    text(currentAx, (ev.onTimes(t) + ev.offTimes(t))/2, max(ylim(currentAx))*0.92, ev.name, 'Color', ev.color, 'FontSize', 8, 'FontWeight', 'bold', 'HorizontalAlignment', 'center', 'Clipping', 'on');
                end
            end
            
            % UPGRADED: Dynamic Figure-Per-Event Subplot Replication Grid Engine
            if ev.isolate && ~isempty(plots)
                numSegments = length(ev.onTimes);
                for s = 1:numSegments
                    % Create a distinct window for each separate event burst window
                    activeBurstFigNum = 200 + (k * 10) + s;
                    zoomFig = figure(activeBurstFigNum); clf(zoomFig);
                    set(zoomFig, 'Name', sprintf('%s - Burst Set Window #%d', ev.name, s));
                    
                    % Replicate the layout structure completely
                    for pIdx = 1:length(plots)
                        pRef = plots{pIdx};
                        zAx = subplot(pRef.row, pRef.col, pRef.pos, 'Parent', zoomFig); hold(zAx, 'on');
                        
                        xVal = getTimelineData(pRef.x, pRef.xCol);
                        for j = 1:length(pRef.y)
                            yN = pRef.y{j}; yV = getProcessedData(yN, pRef.yCol);
                            len = min(length(xVal), length(yV));
                            plot(zAx, xVal(1:len), yV(1:len), 'LineWidth', 1.5, 'DisplayName', dataVars.(yN).label);
                        end
                        
                        % Inject local shading limits and time adjustments
                        xregion(zAx, ev.onTimes(s), ev.offTimes(s), 'FaceColor', ev.color, 'FaceAlpha', ev.opacity, 'HandleVisibility','off');
                        xlim(zAx, [ev.onTimes(s) - ev.buffer, ev.offTimes(s) + ev.buffer]);
                        grid(zAx, 'on'); legend(zAx, 'show');
                    end
                end
            end
        end
        
        % Apply Group Linking Across Arrays
        activeGroups = keys(linkMap);
        for g = 1:length(activeGroups)
            axesArray = linkMap(activeGroups{g}); if length(axesArray) > 1, linkaxes(axesArray, 'x'); end
        end
    end

%% --- FEATURE: FIXED STANDALONE EXECUTABLE M-FILE GENERATOR ---
    function exportToScript(~,~)
        [file, path] = uiputfile('*.m', 'Generate Analytics Plotting Script', 'plot_telemetry_script.m');
        if isequal(file,0), return; end
        fid = fopen(fullfile(path, file), 'w');
        
        fprintf(fid, '%%%% Standalone Automated Analytical Visualization Processing Manifest\n');
        fprintf(fid, '%% Generated on %s\n\n', datestr(now));
        fprintf(fid, 'clearvars; clc;\n\n%% 1. LOAD DATASET\nmatFileName = ''%s'';\n', loadedFileName);
        fprintf(fid, 'if exist(matFileName, ''file''), load(matFileName); else, error(''File missing.''); end\n\n');
        
        fprintf(fid, '%% 2. TRANSLATE & STRIP STRUCT WRAPPERS\n');

        fn = fieldnames(dataVars);

        for i = 1:length(fn)

            internalName = fn{i};

            sourcePath = dataVars.(internalName).sourcePath;

            exportName = matlab.lang.makeValidName(sourcePath);

            fprintf(fid, 'raw_%s = %s;\n', exportName, sourcePath);

            if dataVars.(internalName).transpose
                fprintf(fid, 'raw_%s = raw_%s'';\n', exportName, exportName);
            end

            fprintf(fid, '\n');

        end
        
        fprintf(fid, '%% 3. BUILD RENDERS AND LAYOUT SUBPLOTS\n');
        allFigs = []; if ~isempty(plots), allFigs = [allFigs, cellfun(@(x) x.figNum, plots)]; end
        uniqueFigs = unique(allFigs);
        
        for f = uniqueFigs
            fprintf(fid, 'figure(%d); clf; clear ax;\n', f);
            for k = 1:length(plots)
                p = plots{k}; if p.figNum ~= f, continue; end
                safeX = strrep(p.x, '.', '_');
                fprintf(fid, 'ax(%d) = subplot(%d, %d, %d); hold(ax(%d), ''on'');\n', p.pos, p.row, p.col, p.pos, p.pos);
                fprintf(fid, 'set(ax(%d), ''UserData'', %d);\n', p.pos, p.pos);
                fprintf(fid, '  if size(raw_%s, 2) >= %d, t_vec_%d = double(raw_%s(:, %d)); else, t_vec_%d = (1:length(raw_%s))''; end\n', safeX, p.xCol, k, safeX, p.xCol, k, safeX);

                if p.isCustom

                    fprintf(fid, '  try\n');

                    for j = 1:length(fn)

                        srcPath = dataVars.(fn{j}).sourcePath;
                        sV = matlab.lang.makeValidName(srcPath);

                        fprintf(fid, ...

                        p.pos, k,p.customName);

                            fprintf(fid, '      xlim(zAx, [onTimes_%d(s) - %g, offTimes_%d(s) + %g]);\n', k, ev.buffer, k, ev.buffer);
                            fprintf(fid, '      grid(zAx, ''on''); legend(zAx, ''show'');\n');
                        end
                        fprintf(fid, '  end\n\n');
                    end
                end
            end
        end
        
        if ~isempty(plots)
            fprintf(fid, '\n%% 4. SELECTIVE AXIS GROUP LINKING\n');
            allGroups = cellfun(@(x) x.linkGroup, plots); uGroups = unique(allGroups(allGroups > 0));
            for g = uGroups

                for k = 1:length(plots)
                    if plots{k}.linkGroup == g

                        fprintf(fid, 'if ~isempty(axTemp), axGroup_%d = [axGroup_%d, axTemp(1)]; end\n', g, g);
                    end
                end
                fprintf(fid, 'if length(axGroup_%d) > 1, linkaxes(axGroup_%d, ''x''); end\n\n', g, g);
            end
        end
        
        fclose(fid); uialert(fig, 'M-File Generated successfully with persistent multi-subplot segmentation grids!', 'Export Complete');
    end

%% --- MAT STRUCT RESOLUTION HANDLERS ---
    function loadMat(~,~)
    [file,path] = uigetfile('*.mat','Select Dataset File');
    if isequal(file,0), return; end

    loadedFileName = file;
    dataVars = struct;

    S = load(fullfile(path,file));

    % Recursively scan everything
    dataVars = scanStructRecursive(S, '');

    refreshTable();
    updateSignalLists();
    end

function outStruct = scanStructRecursive(inStruct, prefix)

    outStruct = struct;

    fields = fieldnames(inStruct);

    for i = 1:length(fields)

        fname = fields{i};
        value = inStruct.(fname);

        % Build hierarchical name
        if isempty(prefix)
            fullName = fname;
        else
            fullName = [prefix '.' fname];
        end

        % Safe MATLAB struct field
        safeName = matlab.lang.makeValidName(fullName);

        if isstruct(value)

            tempStruct = scanStructRecursive(value, fullName);

            tf = fieldnames(tempStruct);

            for k = 1:length(tf)
                outStruct.(tf{k}) = tempStruct.(tf{k});
            end

        else

            outStruct.(safeName) = struct( ...
                'data', value, ...
                'sourcePath', fullName, ...
                'label', fname, ...
                'unit', '', ...
                'scale', 1, ...
                'offset', 0, ...
                'transpose', false, ...
                'style', '-', ...
                'width', 1.5);

        end
    end
end

    function refreshTable()
        n = fieldnames(dataVars); d = cell(length(n),8);
        for i = 1:length(n), v = dataVars.(n{i}); d(i,:) = { v.label,v.label, v.unit, v.scale, v.offset, v.transpose, v.style, v.width}; end
        table.Data = d;
    end

    function updateSignalLists()

    n = fieldnames(dataVars);

    labels = cell(size(n));

    for i = 1:length(n)
        labels{i} = dataVars.(n{i}).label;
    end

    xList.Items = labels;
    yList.Items = labels;

    % IMPORTANT:
    xList.ItemsData = n;
    yList.ItemsData = n;
end

    function tableEdit(~,~), d = table.Data; for i = 1:size(d,1), safeFields = fieldnames(dataVars);
n = safeFields{i}; dataVars.(n).label=d{i,2}; dataVars.(n).unit=d{i,3}; dataVars.(n).scale=d{i,4}; dataVars.(n).offset=d{i,5}; dataVars.(n).transpose=d{i,6}; dataVars.(n).style=d{i,7}; dataVars.(n).width=d{i,8}; end; end
   
    function filterX(~,~)

    n = fieldnames(dataVars);

    labels = {};
    vals = {};

    for i = 1:length(n)

        lbl = dataVars.(n{i}).label;

        if contains(lower(lbl), lower(xSearch.Value))
            labels{end+1} = lbl;
            vals{end+1} = n{i};
        end
    end

    xList.Items = labels;
    xList.ItemsData = vals;
    end

function filterY(~,~)

    n = fieldnames(dataVars);

    labels = {};
    vals = {};

    for i = 1:length(n)

        lbl = dataVars.(n{i}).label;

        if contains(lower(lbl), lower(ySearch.Value))
            labels{end+1} = lbl;
            vals{end+1} = n{i};
        end

    yList.Items = labels;

end

    function applyOffset(~,~), o = table.Data{1,5}; ns = fieldnames(dataVars); for i=1:length(ns), dataVars.(ns{i}).offset = o; end; refreshTable(); end
    function clearPlots(~,~), plots={}; appEvents={}; layoutMapping=[]; plotList.Items={}; clf; end
    function searchVar(~,~), k = lower(searchBox.Value); ns = fieldnames(dataVars); tD = {}; for i=1:length(ns), if contains(lower(ns{i}),k), v=dataVars.(ns{i}); tD(end+1,:)={ns{i},v.label,v.unit,v.scale,v.offset,v.transpose,v.style,v.width}; end; end; table.Data=tD; end

function mask = evaluateLogicalExpression(exprStr, wsStruct)
    fn = fieldnames(wsStruct);
    for idx = 1:length(fn)
        eval([fn{idx} ' = wsStruct.' fn{idx} ';']);
    end
    mask = eval(exprStr);
end

function addPrefixToLabels(~,~)

    answer = inputdlg('Enter prefix:','Prefix',1,{'SIM_'});

    if isempty(answer)
        return;
    end

    prefix = answer{1};

    selectedRows = table.Selection;

    if isempty(selectedRows)

        uialert(fig,...
            'Select rows in table first.',...
            'No Selection');

        return;
    end

    rows = unique(selectedRows(:,1));

    d = table.Data;

    for i = 1:length(rows)

        r = rows(i);

        d{r,2} = [prefix d{r,2}];

    end

    table.Data = d;

    tableEdit();

end
