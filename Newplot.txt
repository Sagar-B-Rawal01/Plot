    %% ====================================================================

        function obj = TelemetryWorkbench()
            obj.DataVars = struct();
            obj.Plots = {};
            obj.Events = {};
            obj.LoadedFiles = {};
            obj.SelectedRows = [];
            obj.EvalStack = {};
            obj.ExtState = struct('startX', NaN, 'endX', NaN, 'startIdx', NaN, 'endIdx', NaN, 'hStart', [], 'hEnd', []);
            obj.UI = struct();

            obj.buildApplicationUI();
            obj.logMessage('SYSTEM', 'Modular Telemetry Workbench initialized. Ready for data.');
        end

        function delete(obj)
            % Clean up the UI figure if the object is deleted
            if isvalid(obj) && isfield(obj.UI, 'Fig') && isvalid(obj.UI.Fig)
                delete(obj.UI.Fig);
            end
        end
    end

    %% ====================================================================
    %% 5. ERROR HANDLING & LOGGING (DEBUGGABILITY)
    %% ====================================================================
    methods (Access = private)
        function safeExecute(obj, functionHandle, startMsg)
            try
                if ~isempty(startMsg), obj.logMessage('ACTION', startMsg); end
                functionHandle();
            catch ME
                if isempty(ME.stack)
                    errLine = 0;
                    errFile = 'Unknown';
                else
                    errLine = ME.stack(1).line;
                    errFile = ME.stack(1).name;
                end
                errMsg = sprintf('%s (Line %d in %s)', ME.message, errLine, errFile);
                obj.logMessage('ERROR', errMsg);
                fprintf(2, '[DEBUG ERROR] %s\n', errMsg);
                uialert(obj.UI.Fig, errMsg, 'Runtime Error');
            end
        end

        function logMessage(obj, level, msg)
            txt = sprintf('[%s] [%s] %s', datestr(now, 'HH:MM:SS'), level, msg);
            obj.UI.Console.Value{end+1} = txt;
            scroll(obj.UI.Console, 'bottom');
        end
    end

    methods (Access = private)

        % --- File Management ---
        function actionLoadMat(obj)
            [file,path] = uigetfile('*.mat', 'Select MAT Files', 'MultiSelect','on');
            if isequal(file,0), return; end
            if ischar(file), file = {file}; end

            for k = 1:numel(file)
                fullFile = fullfile(path,file{k});
                obj.LoadedFile = file{k};
                obj.LoadedFiles{end+1} = fullFile;
                rawStruct = load(fullFile);
                obj.parseStructRecursive(rawStruct,'');
                obj.logMessage('LOAD', sprintf('Loaded %s',file{k}));
            end
            obj.actionRefreshTable();
            obj.syncUILists();
        end

        function parseStructRecursive(obj,inStruct,prefix)
            fn = fieldnames(inStruct);
            for i = 1:length(fn)
                val = inStruct.(fn{i});
                if isempty(prefix)
                    fullPath = fn{i};
                else
                    fullPath = [prefix '.' fn{i}];
                end

                if isstruct(val)
                    obj.parseStructRecursive(val,fullPath);
                else
                    if isnumeric(val) && size(val,1)>1 && size(val,2)>1
                        for c = 1:size(val,2)
                            colPath = sprintf('%s_col%d',fullPath,c);
                            safeName = matlab.lang.makeValidName(colPath);
                            safeName = obj.makeUniqueName(safeName);
                            fileTag = erase(obj.LoadedFile,'.mat');
                            sourcePath = sprintf('%s.%s_col%d', fileTag, fullPath, c);
                            tokens = split(fullPath,'.');
                            label = char(sprintf('%s_%d', tokens{end}, c));
                            
                            obj.DataVars.(safeName) = obj.createMetaStruct(val(:,c), sourcePath, label);
                        end
                    else
                        safeName = matlab.lang.makeValidName(fullPath);
                        safeName = obj.makeUniqueName(safeName);
                        fileTag = erase(obj.LoadedFile,'.mat');
                        sourcePath = sprintf('%s.%s', fileTag, fullPath);
                        tokens = split(fullPath,'.');
                        label = char(tokens{end});
                        
                        obj.DataVars.(safeName) = obj.createMetaStruct(val, sourcePath, label);
                    end
                end
            end
        end

        function meta = createMetaStruct(~, data, sourcePath, label)
            meta = struct('data', data, 'sourcePath', sourcePath, 'label', char(label), ...
                'unit', '', 'scale', 1, 'offset', 0, 'transpose', false, ...
                'style', '-', 'width', 1.5, 'color', 'auto', 'isDerived', false, ...
                'expr', '', 'cache', []);
        end

        % --- Math / Signal Processing Engine ---
        function sig = fetchSignal(obj, key)
            meta = obj.DataVars.(key);

            if meta.isDerived
                if ismember(key, obj.EvalStack)
                    chain = strjoin(obj.EvalStack, ' -> ');
                    error('Cyclic dependency detected: %s -> %s', chain, key);
                end
                obj.EvalStack{end+1} = key;
                try
                    sig = obj.evaluateMathSafely(meta.expr);
                    obj.DataVars.(key).cache = sig;
                    obj.EvalStack(end) = [];
                catch ME
                    obj.EvalStack = {}; rethrow(ME);
                end
                return;
            end

            if ~isempty(meta.cache), sig = meta.cache; return; end

            raw = meta.data;
            if istable(raw), raw = table2array(raw); end
            if isrow(raw), raw = raw(:); end
            if meta.transpose, raw = raw'; end
            raw = double(raw);
            if size(raw, 2) > 1, raw = raw(:,1); end

            sig = raw * meta.scale + meta.offset;
            sig(~isfinite(sig)) = NaN;
            obj.DataVars.(key).cache = sig;
        end

        function out = evaluateMathSafely(obj, expr)
            fn = fieldnames(obj.DataVars);
            for i = 1:length(fn)
                meta = obj.DataVars.(fn{i});
                if meta.isDerived
                    if ismember(fn{i}, obj.EvalStack), continue; end
                    try val = obj.fetchSignal(fn{i}); catch, continue; end
                else
                    raw = meta.data;
                    if istable(raw), raw = table2array(raw); end
                    if isrow(raw), raw = raw(:); end
                    if meta.transpose, raw = raw'; end
                    raw = double(raw);
                    if size(raw,2) > 1, raw = raw(:,1); end
                    val = raw * meta.scale + meta.offset;
                    val(~isfinite(val)) = NaN;
                end
                try eval([meta.sourcePath ' = val;']); catch; end %#ok<CTCH>
            end
            out = eval(expr);
        end

        function actionCreateDerived(obj)
            name = char(strtrim(obj.UI.derName.Value));
            expr = char(strtrim(obj.UI.derExpr.Value));
            if isempty(name) || isempty(expr), error('Name and Expression required.'); end
            safeName = matlab.lang.makeValidName(name);
            out = obj.evaluateMathSafely(expr);
            if ~isnumeric(out), error('Expression must result in a numeric array.'); end

            obj.DataVars.(safeName) = struct('data', out(:), 'sourcePath', safeName, 'label', name, ...
                'unit', char(obj.UI.derUnit.Value), 'scale', 1, 'offset', 0, 'transpose', false, ...
                'style', '-', 'width', 1.5, 'color', 'auto', 'isDerived', true, 'expr', expr, 'cache', out(:));

            obj.actionRefreshTable(); obj.syncUILists();
        end

        % --- UI State Synchronization ---
        function actionRefreshTable(obj)
            k = lower(char(obj.UI.txtSearch.Value));
            fn = fieldnames(obj.DataVars);
            d = cell(0, 11); rowIdx = 1;
            for i = 1:length(fn)
                v = obj.DataVars.(fn{i});
                if isempty(k) || contains(lower(v.label), k) || contains(lower(fn{i}), k)
                    try
                        sz = size(v.data);
                        if v.transpose, sz = fliplr(sz); end
                        szTxt = sprintf('%dx%d', sz(1), sz(2));
                    catch
                        szTxt = 'Unknown';
                    end
                    d(rowIdx, :) = {fn{i}, v.label, v.unit, szTxt, v.scale, v.offset, v.transpose, v.style, v.width, v.color, v.isDerived};
                    rowIdx = rowIdx + 1;
                end
            end
            obj.UI.dataTable.Data = d;
        end

        function syncUILists(obj)
            fn = fieldnames(obj.DataVars);
            lbls = cell(size(fn));
            for i=1:length(fn), lbls{i} = char(obj.DataVars.(fn{i}).label); end

            obj.UI.pltXList.Items = lbls; obj.UI.pltXList.ItemsData = fn;
            obj.UI.pltYList.Items = lbls; obj.UI.pltYList.ItemsData = fn;
            obj.UI.extX.Items = lbls; obj.UI.extX.ItemsData = fn;
            obj.UI.extY.Items = lbls; obj.UI.extY.ItemsData = fn;
            obj.UI.extExportList.Items = lbls; obj.UI.extExportList.ItemsData = fn;
        end

        function actionFilterList(obj, axis)
            if strcmp(axis, 'x')
                txt = lower(char(obj.UI.pltXSearch.Value)); list = obj.UI.pltXList;
            else
                txt = lower(char(obj.UI.pltYSearch.Value)); list = obj.UI.pltYList;
            end
            fn = fieldnames(obj.DataVars); matches = {}; lbls = {};
            for i = 1:length(fn)
                lbl = char(obj.DataVars.(fn{i}).label);
                if isempty(txt) || contains(lower(lbl), txt) || contains(lower(fn{i}), txt)
                    matches{end+1} = fn{i}; lbls{end+1} = lbl; %#ok<AGROW>
                end
            end
            list.Items = lbls; list.ItemsData = matches;
        end

        function syncManifest(obj)
            items = {}; ids = {};
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                if p.isCus
                    txt = sprintf('[Plot] F%d(S%d) %s', p.f, p.p, p.name);
                else
                    txt = sprintf('[Plot] F%d(S%d) %s', p.f, p.p, strjoin(p.y, ',')); 
                end
                items{end+1} = txt; ids{end+1} = p.id; %#ok<AGROW>
            end
            for i = 1:length(obj.Events)
                e = obj.Events{i};
                items{end+1} = sprintf('[Event] F%d %s', e.fig, e.name); ids{end+1} = e.id; %#ok<AGROW>
            end
            obj.UI.plotManifest.Items = items;
            obj.UI.plotManifest.ItemsData = ids;
            obj.UI.plotManifest.Value = {};
        end

        % --- Table Interaction ---
        function actionTableSelect(obj, ev)
            if isempty(ev.Indices), obj.SelectedRows = []; return; end
            obj.SelectedRows = unique(ev.Indices(:,1))';
        end

        function actionTableEdit(obj, ev)
            r = ev.Indices(1); c = ev.Indices(2);
            key = obj.UI.dataTable.Data{r, 1}; val = ev.NewData;
            switch c
                case 2, obj.DataVars.(key).label = char(val); obj.syncUILists(); 
                case 3, obj.DataVars.(key).unit = char(val);
                case 5, obj.DataVars.(key).scale = val; obj.DataVars.(key).cache = [];
                case 6, obj.DataVars.(key).offset = val; obj.DataVars.(key).cache = [];
                case 7, obj.DataVars.(key).transpose = val; obj.DataVars.(key).cache = [];
                case 8, obj.DataVars.(key).style = char(val);
                case 9, obj.DataVars.(key).width = val;
                case 10, obj.DataVars.(key).color = char(val);
            end
            obj.renderAll();
        end

        function keys = getSelectedKeys(obj)
            keys = {}; if isempty(obj.SelectedRows), return; end
            for i = 1:length(obj.SelectedRows)
                r = obj.SelectedRows(i);
                if r <= size(obj.UI.dataTable.Data, 1), keys{end+1} = obj.UI.dataTable.Data{r, 1}; end %#ok<AGROW>
            end
        end

        function actionApplyModifier(obj, mode)
            keys = obj.getSelectedKeys(); if isempty(keys), error('Select rows in the table first.'); end
            val = inputdlg(['Enter numerical ' mode ':']); if isempty(val), return; end
            num = str2double(val{1}); if isnan(num), error('Invalid number.'); end
            for i = 1:length(keys)
                obj.DataVars.(keys{i}).(mode) = num; obj.DataVars.(keys{i}).cache = [];
            end
            obj.actionRefreshTable(); obj.renderAll();
        end

        function actionApplyPrefix(obj)
            keys = obj.getSelectedKeys(); if isempty(keys), error('Select rows in the table first.'); end
            val = inputdlg('Enter Prefix:'); if isempty(val), return; end
            for i = 1:length(keys)
                obj.DataVars.(keys{i}).label = [char(val{1}) char(obj.DataVars.(keys{i}).label)];
            end
            obj.actionRefreshTable(); obj.syncUILists();
        end

        function actionResetModifiers(obj)
            keys = obj.getSelectedKeys(); if isempty(keys), keys = fieldnames(obj.DataVars); end
            for i = 1:length(keys)
                obj.DataVars.(keys{i}).scale = 1; obj.DataVars.(keys{i}).offset = 0; obj.DataVars.(keys{i}).cache = [];
            end
            obj.actionRefreshTable(); obj.renderAll();
        end

            if pNum > (r * c), error('Subplot position exceeds Matrix size (Rows x Cols).'); end

            if isCustom
                p = struct('id', id, 'isCus', true, 'xExp', char(obj.UI.cusX.Value), 'yExp', char(obj.UI.cusY.Value), ...
                    'name', char(obj.UI.cusLbl.Value), 'f', obj.UI.pltFig.Value, 'r', r, 'c', c, 'p', pNum, ...
                    'type', char(obj.UI.pltType.Value), 'grid', obj.UI.pltGrid.Value, 'leg', obj.UI.pltLeg.Value, ...
                    'title', char(obj.UI.pltTitle.Value), 'link', obj.UI.pltLink.Value);
            else
                if isempty(obj.UI.pltXList.Value) || isempty(obj.UI.pltYList.Value), error('Select X and Y signals.'); end
                
                % CRITICAL FIX: Wrapped cellstr inside curly braces to prevent struct array generation
                p = struct('id', id, 'isCus', false, 'x', char(obj.UI.pltXList.Value), 'y', {cellstr(obj.UI.pltYList.Value)}, ...
                    'f', obj.UI.pltFig.Value, 'r', r, 'c', c, 'p', pNum, ...
                    'type', char(obj.UI.pltType.Value), 'grid', obj.UI.pltGrid.Value, 'leg', obj.UI.pltLeg.Value, ...
                    'title', char(obj.UI.pltTitle.Value), 'link', obj.UI.pltLink.Value);
            end

            obj.Plots{end+1} = p;
            obj.syncManifest(); obj.renderAll();
        end

        function actionDeleteManifestItem(obj)
            selId = obj.UI.plotManifest.Value;
            if isempty(selId), return; end
            selId = char(selId);
            obj.Plots = obj.Plots(~cellfun(@(x) strcmp(x.id, selId), obj.Plots));
            obj.Events = obj.Events(~cellfun(@(x) strcmp(x.id, selId), obj.Events));
            obj.syncManifest(); obj.renderAll();
        end

        function actionAddEvent(obj)
            expr = char(strtrim(obj.UI.evLogic.Value)); if isempty(expr), error('Event logic cannot be empty.'); end
            mask = obj.evaluateMathSafely(expr); if isempty(mask), error('Event expression returned empty.'); end
            mask = double(mask(:)); mask(~isfinite(mask)) = 0; mask(mask ~= 0) = 1;

            d = diff([0; mask; 0]); startIdx = find(d == 1); stopIdx = find(d == -1);
            stopIdx(stopIdx > length(mask)) = length(mask);
            if isempty(startIdx), error('No event transitions detected.'); end

            fn = fieldnames(obj.DataVars); labels = cell(size(fn));
            for i = 1:length(fn), labels{i} = char(obj.DataVars.(fn{i}).label); end
            [indx, tf] = listdlg('PromptString', 'Select X-axis signal for event timeline', 'SelectionMode', 'single', 'ListString', labels);
            if ~tf, return; end

            t = obj.fetchSignal(fn{indx}); t = t(:);
            if length(t) < length(mask), error('X-axis signal shorter than event mask.'); end

            e = struct('id', char(javaMethod('randomUUID','java.util.UUID')), 'name', char(obj.UI.evName.Value), 'logic', expr, ...
                'onT', t(startIdx), 'offT', t(stopIdx), 'fig', obj.UI.evFig.Value, 'c', char(obj.UI.evColor.Value), 'alpha', obj.UI.evOpac.Value);
            obj.Events{end+1} = e;
            obj.logMessage('EVENT', sprintf('%s -> %d ON transitions detected.', e.name, length(startIdx)));
            obj.syncManifest(); obj.renderAll();
        end

        % --- Core Rendering Engine ---
        function renderAll(obj)
            plotFigs = [];
            eventFigs = [];

            if ~isempty(obj.Plots), plotFigs = cellfun(@(x)x.f,obj.Plots); end
            if ~isempty(obj.Events), eventFigs = cellfun(@(x)x.fig,obj.Events); end

            allFigs = unique([plotFigs eventFigs]);

            if isempty(allFigs), return; end
            for i = 1:length(allFigs), figure(allFigs(i)); clf(figure(allFigs(i))); end

            axMap = containers.Map('KeyType', 'double', 'ValueType', 'any'); % To store axes for linking

            % Render Plots
            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                ax = subplot(p.r, p.c, p.p, 'Parent', figure(p.f)); hold(ax, 'on');

                % Store axis for linking if ID > 0
                if p.link > 0
                    if isKey(axMap, p.link)
                        axMap(p.link) = [axMap(p.link), ax];
                    else
                        axMap(p.link) = ax;
                    end
                end

                if p.isCus
                    xV = obj.evaluateMathSafely(p.xExp); yV = obj.evaluateMathSafely(p.yExp);
                    plot(ax, xV, yV, 'LineWidth', 1.5, 'DisplayName', p.name);
                    xlabel(ax, p.xExp, 'Interpreter', 'none'); ylabel(ax, p.yExp, 'Interpreter', 'none');
                else
                    xV = obj.fetchSignal(p.x); lbls = {};
                    for j = 1:length(p.y)
                        yV = obj.fetchSignal(p.y{j}); meta = obj.DataVars.(p.y{j});
                        len = min(length(xV), length(yV));

                        args = {'LineWidth', meta.width, 'LineStyle', meta.style, 'DisplayName', char(meta.label)};
                        if ~strcmp(meta.color, 'auto'), args = [args, 'Color', meta.color]; end
                        feval(p.type, ax, xV(1:len), yV(1:len), args{:});

                        lbls{end+1} = char(meta.label); %#ok<AGROW>
                    end
                    xlabel(ax, char(obj.DataVars.(p.x).label), 'Interpreter', 'none');

                    % SMART Y-LABEL
                    ylabel(ax, strjoin(lbls, ' / '), 'Interpreter', 'none');
                end

                if ~isempty(p.title), title(ax, p.title, 'Interpreter', 'none'); end
                if p.grid, grid(ax, 'on'); end
                if p.leg, legend(ax, 'Location', 'best', 'Interpreter', 'none'); end
            end

            % Process Link Groups
            linkIds = axMap.keys();
            for i = 1:length(linkIds)
                axList = axMap(linkIds{i});
                if length(axList) > 1
                    linkaxes(axList, 'x'); % Link X axes for the grouped subplots
                end
            end

            % Render Events
            for i = 1:length(obj.Events)
                e = obj.Events{i};
                axesList = findall(figure(e.fig), 'Type', 'axes', '-not', 'Tag', 'legend', '-not', 'Tag', 'Colorbar');
                for a = 1:length(axesList)
                    ax = axesList(a);
                    delete(findobj(ax, 'Tag', 'eventOverlay'));
                    yl = ylim(ax); hold(ax, 'on');
                    for t = 1:length(e.onT)
                        patch(ax, [e.onT(t) e.offT(t) e.offT(t) e.onT(t)], [yl(1) yl(1) yl(2) yl(2)], e.c, 'FaceAlpha', e.alpha, 'EdgeColor', 'none', 'Tag', 'eventOverlay', 'HandleVisibility', 'off');
                    end
                end
            end
        end

        % --- Script Export ---
        function actionExportScript(obj)
            [file, path] = uiputfile('*.m', 'Save Script', 'plot_telemetry.m');
            if file == 0, return; end
            fid = fopen(fullfile(path, file), 'w');

            fprintf(fid, '%% STANDALONE TELEMETRY DASHBOARD SCRIPT\n');
            fprintf(fid, '%% Generated by TelemetryWorkbench\n');
            fprintf(fid, 'clearvars; clc;\n\n');

            fprintf(fid, '%% --- 1. Load Data ---\n');
            for f = 1:length(obj.LoadedFiles)
                fprintf(fid,'load(''%s'');\n',strrep(obj.LoadedFiles{f},'''', ''''''));
            end
            fprintf(fid, '\n');

            fprintf(fid, '%% --- 2. Format Raw Data ---\n');
            fn = fieldnames(obj.DataVars);
            for i = 1:length(fn)
                m = obj.DataVars.(fn{i});
                if ~m.isDerived
                    fprintf(fid, '%s = double(%s);\n', m.sourcePath, m.sourcePath);
                    if m.transpose, fprintf(fid, 'if size(%s, 1) < size(%s, 2), %s = %s''; end\n', m.sourcePath, m.sourcePath, m.sourcePath, m.sourcePath); end
                    fprintf(fid, 'if isrow(%s), %s = %s(:); end\n', m.sourcePath, m.sourcePath, m.sourcePath);
                    fprintf(fid, 'if size(%s,2) > 1, %s = %s(:,1); end\n', m.sourcePath, m.sourcePath, m.sourcePath);
                    fprintf(fid, '%s = %s * %g + %g;\n\n', m.sourcePath, m.sourcePath, m.scale, m.offset);
                end
            end

            fprintf(fid, '%% --- 3. Compute Derived Signals ---\n');
            for i = 1:length(fn)
                m = obj.DataVars.(fn{i});
                if m.isDerived, fprintf(fid, '%s = %s;\n', m.sourcePath, m.expr); end
            end

            fprintf(fid, '\n%% --- 4. Generate Plots ---\n');
            fprintf(fid, 'axMap = containers.Map(''KeyType'', ''double'', ''ValueType'', ''any'');\n\n');

            for i = 1:length(obj.Plots)
                p = obj.Plots{i};
                fprintf(fid, '%% Subplot F%d (Row %d, Col %d, Pos %d)\n', p.f, p.r, p.c, p.p);
                fprintf(fid, 'figure(%d);\nax = subplot(%d, %d, %d);\nhold(ax, ''on'');\n', p.f, p.r, p.c, p.p);

                % Store axis handle for linked groups
                if p.link > 0
                    fprintf(fid, 'if isKey(axMap, %d), axMap(%d) = [axMap(%d), ax]; else, axMap(%d) = ax; end\n', p.link, p.link, p.link, p.link);
                end

                if p.isCus
                    fprintf(fid, 'plot(ax, %s, %s, ''LineWidth'', 1.5, ''DisplayName'', ''%s'');\n', p.xExp, p.yExp, p.name);
                    fprintf(fid, 'xlabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(p.xExp, '''', ''''''));
                    fprintf(fid, 'ylabel(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(p.yExp, '''', ''''''));
                else
                    lbls = {};
                    for j = 1:length(p.y)
                        m = obj.DataVars.(p.y{j});
                        fprintf(fid, 'plot(ax, %s, %s, ''LineWidth'', %g, ''DisplayName'', ''%s'');\n', ...
                            obj.DataVars.(p.x).sourcePath, m.sourcePath, m.width, char(m.label));
                        lbls{end+1} = char(m.label); %#ok<AGROW>
                    end
                    fprintf(fid, 'xlabel(ax, ''%s'', ''Interpreter'', ''none'');\n', char(obj.DataVars.(p.x).label));
                    safeYLabel = strrep(strjoin(lbls, ' / '), '''', '''''');
                    fprintf(fid, 'ylabel(ax, ''%s'', ''Interpreter'', ''none'');\n', safeYLabel);
                end

                if ~isempty(p.title), fprintf(fid, 'title(ax, ''%s'', ''Interpreter'', ''none'');\n', strrep(p.title, '''', '''''')); end
                if p.grid, fprintf(fid, 'grid(ax, ''on'');\n'); end
                if p.leg, fprintf(fid, 'legend(ax, ''Location'', ''best'', ''Interpreter'', ''none'');\n'); end
                fprintf(fid, '\n');
            end

            fprintf(fid, '%% --- 5. Apply Axis Linking ---\n');
            fprintf(fid, 'linkIds = axMap.keys();\n');
            fprintf(fid, 'for i = 1:length(linkIds)\n');
            fprintf(fid, '    axList = axMap(linkIds{i});\n');
            fprintf(fid, '    if length(axList) > 1, linkaxes(axList, ''x''); end\n');
            fprintf(fid, 'end\n\n');

            fprintf(fid, '%% --- 6. Event Logic Overlay ---\n');
            for i = 1:length(obj.Events)
                e = obj.Events{i};
                fprintf(fid, 'figure(%d);\n', e.fig);
                fprintf(fid, 'eventOn = [%s];\n', num2str(e.onT'));
                fprintf(fid, 'eventOff = [%s];\n', num2str(e.offT'));
                fprintf(fid, 'axList = findall(gcf, ''Type'', ''axes'', ''-not'', ''Tag'', ''legend'');\n');
                fprintf(fid, 'for ax_i = 1:length(axList)\n');
                fprintf(fid, '    ax = axList(ax_i); hold(ax, ''on''); yl = ylim(ax);\n');
                fprintf(fid, '    for k = 1:length(eventOn)\n');
                fprintf(fid, '        xline(ax, eventOn(k), ''-'', ''ON : %s'', ''Color'', ''g'', ''LineWidth'', 1.5);\n', e.name);
                fprintf(fid, '        xline(ax, eventOff(k), ''-'', ''OFF : %s'', ''Color'', ''r'', ''LineWidth'', 1.5);\n', e.name);
                fprintf(fid, '        patch(ax, [eventOn(k) eventOff(k) eventOff(k) eventOn(k)], [yl(1) yl(1) yl(2) yl(2)], ''%s'', ''FaceAlpha'', %g, ''EdgeColor'', ''none'', ''HandleVisibility'', ''off'');\n', e.c, e.alpha);
                fprintf(fid, '    end\nend\n');
            end
            fclose(fid);
        end

        % --- Extraction Tool (Unchanged) ---
        function actionRenderExtract(obj)
            xKey = char(obj.UI.extX.Value); yKey = char(obj.UI.extY.Value);
            if isempty(xKey) || isempty(yKey), error('Select X and Y signals.'); end
            xV = obj.fetchSignal(xKey); yV = obj.fetchSignal(yKey);
            len = min(length(xV), length(yV)); ax = obj.UI.extAxes; cla(ax);
            plot(ax, xV(1:len), yV(1:len), 'ButtonDownFcn', @(~,ev)obj.actionExtractClick(ev));
            grid(ax, 'on'); xlabel(ax, char(obj.DataVars.(xKey).label), 'Interpreter','none'); ylabel(ax, char(obj.DataVars.(yKey).label), 'Interpreter','none');
            obj.ExtState.xData = xV(1:len);
            obj.ExtState.startX = NaN; obj.ExtState.endX = NaN; obj.ExtState.startIdx = NaN; obj.ExtState.endIdx = NaN;
            obj.UI.lblExtStart.Text = 'Start: NaN'; obj.UI.lblExtEnd.Text = 'End: NaN';
        end

        function actionEnableExtCursor(obj)
            obj.logMessage('INFO', 'Click on the plotted blue line to set a marker.');
        end

        function actionExtractClick(obj, ev)
            xClick = ev.IntersectionPoint(1);
            [~, idx] = min(abs(obj.ExtState.xData - xClick)); xSnap = obj.ExtState.xData(idx);

            if isnan(obj.ExtState.startX) || (~isnan(obj.ExtState.startX) && ~isnan(obj.ExtState.endX))
                obj.ExtState.startX = xSnap; obj.ExtState.startIdx = idx;
                obj.ExtState.endX = NaN; obj.ExtState.endIdx = NaN;
                if isgraphics(obj.ExtState.hStart), delete(obj.ExtState.hStart); end
                if isgraphics(obj.ExtState.hEnd), delete(obj.ExtState.hEnd); end
                hold(obj.UI.extAxes, 'on');
                obj.ExtState.hStart = xline(obj.UI.extAxes, xSnap, '-g', 'LineWidth', 2, 'Label', 'Start');
                obj.UI.lblExtStart.Text = sprintf('Start: Idx %d (X=%.2f)', idx, xSnap);
                obj.UI.lblExtEnd.Text = 'End: NaN';
            else
                obj.ExtState.endX = xSnap; obj.ExtState.endIdx = idx;
                if obj.ExtState.endIdx < obj.ExtState.startIdx
                    [obj.ExtState.startX, obj.ExtState.endX] = deal(obj.ExtState.endX, obj.ExtState.startX);
                    [obj.ExtState.startIdx, obj.ExtState.endIdx] = deal(obj.ExtState.endIdx, obj.ExtState.startIdx);
                    delete(obj.ExtState.hStart);
                    obj.ExtState.hStart = xline(obj.UI.extAxes, obj.ExtState.startX, '-g', 'LineWidth', 2, 'Label', 'Start');
                end
                if isgraphics(obj.ExtState.hEnd), delete(obj.ExtState.hEnd); end
                obj.ExtState.hEnd = xline(obj.UI.extAxes, obj.ExtState.endX, '-r', 'LineWidth', 2, 'Label', 'End');
                obj.UI.lblExtStart.Text = sprintf('Start: Idx %d (X=%.2f)', obj.ExtState.startIdx, obj.ExtState.startX);
                obj.UI.lblExtEnd.Text = sprintf('End: Idx %d (X=%.2f)', obj.ExtState.endIdx, obj.ExtState.endX);
            end
        end

        function actionExportData(obj)
            if isnan(obj.ExtState.startIdx) || isnan(obj.ExtState.endIdx), error('Set both Start and End markers.'); end
            chans = obj.UI.extExportList.Value; if isempty(chans), error('Select channels to export.'); end
            fName = char(obj.UI.extFile.Value); fid = fopen(fName, 'w'); if fid == -1, error('Failed to create file.'); end
            lbls = cell(1, length(chans)); for c=1:length(chans), lbls{c} = char(obj.DataVars.(chans{c}).label); end
            fprintf(fid, '%s\n', strjoin(lbls, ','));

            for r = obj.ExtState.startIdx : obj.ExtState.endIdx
                rowVals = cell(1, length(chans));
                for c = 1:length(chans)
                    sig = obj.fetchSignal(chans{c});
                    if r <= length(sig), rowVals{c} = sprintf('%.6f', sig(r)); else, rowVals{c} = 'NaN'; end
                end
                fprintf(fid, '%s\n', strjoin(rowVals, ','));
            end
            fclose(fid);
        end

        % --- State Persistence ---
        function actionSaveState(obj)
            [file, path] = uiputfile('*.mat', 'Save State'); if file == 0, return; end
            appState = struct('DataVars', obj.DataVars, 'Plots', {obj.Plots}, 'Events', {obj.Events}, 'LoadedFiles', {obj.LoadedFiles});
            save(fullfile(path, file), 'appState');
        end

        function safeName = makeUniqueName(obj,safeName)
            baseName = safeName;
            count = 1;
            while isfield(obj.DataVars,safeName)
                safeName = matlab.lang.makeValidName(sprintf('%s_%d', baseName, count));
                count = count + 1;
            end
        end

        function actionLoadState(obj)
            [file,path] = uigetfile('*.mat','Load State');
            if isequal(file,0), return; end
            S = load(fullfile(path,file));
            if ~isfield(S,'appState'), error('Invalid state file.'); end

            obj.DataVars = S.appState.DataVars;
            obj.Plots = S.appState.Plots;
            obj.Events = S.appState.Events;
            obj.LoadedFiles = S.appState.LoadedFiles;

            obj.actionRefreshTable();
            obj.syncUILists();
            obj.syncManifest();
            obj.renderAll();
            obj.logMessage('STATE','State loaded successfully.');
        end
